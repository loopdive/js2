function safe(p) {

function checked(root, p, directory = false) {
  safe(p); assert.equal(relative(root, resolve(root,p)),p);
  const absolute = join(root,p), components = absolute.split('/').filter(Boolean);
  let current = '/';
  for (let i=0;i<components.length;i++) {
    current = join(current,components[i]); const stat=fs.lstatSync(current);
    assert(!stat.isSymbolicLink(),current);
    assert(i<components.length-1 || directory ? stat.isDirectory() : stat.isFile(),current);
  }
  return absolute;
}
function raw(root,p) {
  const filename=checked(root,p), fd=fs.openSync(filename,fs.constants.O_RDONLY|fs.constants.O_NOFOLLOW);
  try {
    const a=fs.fstatSync(fd,{bigint:true}), b=fs.readFileSync(fd), z=fs.fstatSync(fd,{bigint:true});
    const leaf=fs.lstatSync(filename,{bigint:true});
    for(const key of ['dev','ino','size','mtimeNs','ctimeNs']) {
      assert.equal(a[key],z[key],`changing input ${p}`); assert.equal(z[key],leaf[key],`replaced input ${p}`);
    }
    assert.equal(BigInt(b.length),z.size); return b;
  } finally {fs.closeSync(fd);}
}
