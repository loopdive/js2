const sha = bytes => createHash('sha256').update(bytes).digest('hex');
function safeRelative(p) {
  assert.equal(typeof p, 'string');
  assert(p.length > 0 && !isAbsolute(p) && !/[\\\x00-\x1f\x7f:]/.test(p), 'Unsafe path: ' + p);
  assert(p.split('/').every(s => s && s !== '.' && s !== '..'), 'Unsafe segments: ' + p);
  assert.equal(relative(E, resolve(E, p)), p);
  return p;
}
function directory(p) {
  assert(isAbsolute(p));
  const ancestors = [];
  for (let at = p; ; at = dirname(at)) {
    ancestors.push(at);
    if (dirname(at) === at) break;
  }
  for (const at of ancestors.reverse()) {
    const st = lstatSync(at);
    assert(st.isDirectory() && !st.isSymbolicLink(), 'Non-directory/symlink: ' + at);
  }
}
function readRegular(p) {
  directory(dirname(p));
  const st = lstatSync(p);
  assert(st.isFile() && !st.isSymbolicLink(), 'Non-regular record: ' + p);
  const fd = openSync(p, constants.O_RDONLY | constants.O_NOFOLLOW);
  try {
    const opened = fstatSync(fd);
    assert(opened.isFile() && opened.dev === st.dev && opened.ino === st.ino, 'Record replaced: ' + p);
    const bytes = readFileSync(fd);
    const after = fstatSync(fd);
    assert.equal(bytes.length, st.size, 'Length drift: ' + p);
    assert.equal(after.size, opened.size);
    assert.equal(after.mtimeMs, opened.mtimeMs);
    assert.equal(after.ctimeMs, opened.ctimeMs);
    return bytes;
  } finally { closeSync(fd); }
}
const captured = new Map();
function capture(p) {
  safeRelative(p);
  assert(!captured.has(p), 'Duplicate path: ' + p);
  const bytes = readRegular(join(E, p));
  if (Object.hasOwn(PINS, p)) assert.equal(sha(bytes), PINS[p], 'Authority drift: ' + p);
  captured.set(p, bytes);
  return bytes;
}

function inventory(p, expected) {
  directory(p);
  assert.deepEqual(readdirSync(p).sort(), [...expected].sort(), 'Unexplained record/directory inventory: ' + p);
}
function stamp(t, label) {
  assert.equal(t.accepted, false, label);
  assert.equal(t.candidateOnly, true, label);
  assert.equal(typeof t.finishedAt, 'string', label);
  assert(Number.isFinite(Date.parse(t.finishedAt)), 'Missing terminal timestamp: ' + label);
}
