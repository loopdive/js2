function safeAbsolute(p) {

function safeRelative(root,p) {
  requireThat(typeof p==='string'&&p.length>0&&!isAbsolute(p)&&!p.includes('\0')&&!p.includes('\\'),'Unsafe relative path');
  requireThat(!p.split('/').some(c=>!c||c==='.'||c==='..'),'Unsafe relative component');

const cache=new Map();
function bytes(p,expected=null,retain=false) {
  let entry=cache.get(p);
  if(!entry) {
    try {
      safeAbsolute(p);
      const b=readFileSync(p);
      entry={b,record:{path:p,bytes:b.length,sha256:sha(b),retained:false,pinChecks:[]}};
    } catch(e) {
      entry={b:null,record:{path:p,unavailable:true,reason:String(e.message??e),retained:false,pinChecks:[]}};
    }
    cache.set(p,entry);report.evidence.push(entry.record);
  }
  if(retain&&!entry.record.retained&&entry.b) {
    entry.record.retained=true;entry.record.originalBytesBase64=entry.b.toString('base64');
  }
  requireThat(entry.b!==null,'Missing/unreadable artifact '+p+': '+entry.record.reason);
  if(expected) {
    const ok=digestRE.test(expected.sha256??'')&&entry.record.sha256===expected.sha256&&(expected.bytes===undefined||entry.record.bytes===expected.bytes);
    entry.record.pinChecks.push({expected,matched:ok});
    requireThat(ok,'Artifact pin mismatch: '+p);
  }
  return entry.b;
}

      }else{

        requireThat(inv.mergeHead===null,'E merge in progress');
        equal(inv.mergeParents,['8b9e89a5f9009d7a17a8c6410fa2ea2b1d0b3960',B_HEAD],'E validated merge parents mismatch');
        equal(inv.identities,Object.fromEntries(stage.files.map(p=>[p,expectedE[p]])),'E invocation identity authority mismatch');
        requireThat(t.code===0&&t.signal===null&&t.error===null,'E child failure');
        equal(t.instrumentationErrors,[],'E instrumentation errors');
      }
      requireThat(t.accepted===false&&t.stage===stage.name&&t.beforeVerified===true&&t.afterVerified===true&&t.stageClean===true,'Stage terminal not clean');
    });
    for(const [key,map] of Object.entries(s.maps))attempt(phase,()=>{
      const a=key.split('-').at(-1);requireThat(map!==undefined,'Missing stage map '+key);
      equal(map,baselines[a],'Stage physical map drift '+stage.name+'/'+key);
    });
    for(const raw of s.raw)attempt(phase,()=>{
      requireThat(typeof raw.text==='string','Missing raw channel '+raw.name);
      requireThat(!rawError.test(raw.text.replace(/\x1b\[[0-9;]*m/g,'')),'Raw worker/RPC/error/FAIL/unhandled channel '+raw.name);
    });
    if(s.nativeRequired)attempt(phase,()=>{requireThat(s.indexed,'Missing complete native rows');evaluateNative(s.native,s.indexed,stage,phase);});
    s.clean=s.available&&s.identityExact&&cleanSince(start,phase);
    // Rows remain available for identity/outcome diagnosis regardless of clean.
  }
}
function rootGates() {
  attempt('B.cleanliness',()=>{
    requireThat(bTerminal&&bTerminal.accepted===false&&bTerminal.arm==='B'&&bTerminal.collectionOnly===false,'Wrong/missing B root terminal');
    requireThat(bTerminal.preflightVerified===true&&bTerminal.rootBeforeVerified===true&&bTerminal.rootAfterVerified===true&&bTerminal.allStagesClean===true&&bTerminal.completedStages===9,'B root incomplete/failed');
    equal(bTerminal.errors,[],'B root errors');equal(bTerminal.attemptedStages,schedule.stages.map(s=>s.name),'B attempted stages mismatch');
    equal(bTerminal.unexecutedStages,[],'B unexecuted stages');
    equal(bTerminal.stageOutcomes,schedule.stages.map(s=>({stage:s.name,stageClean:true})),'B root stage outcomes');
    requireThat(bTerminal.scheduledFiles===70&&bTerminal.scheduledExecutionOccurrences===5510&&bTerminal.sharedBEIdentities===5406&&bTerminal.separateControls===102,'B root denominators');
    for(const arm of ['A','B','E'])for(const side of ['before','after'])equal(parsed(B_RUN+'/'+side+'-'+arm+'.json',null,true),baselines[arm],'B root physical drift '+side+'/'+arm);
  });
  attempt('E.cleanliness',()=>{
    requireThat(eTerminal&&eTerminal.accepted===false&&eTerminal.candidateOnly===true&&eTerminal.allStagesClean===true&&eTerminal.completedStages===10&&eTerminal.expectedStages===10,'E root incomplete/failed');
    requireThat(eTerminal.scheduledUniquePopulation===6739&&eTerminal.scheduledExecutionOccurrences===6795,'E root denominator mismatch');
  });
}
function foldUnique(arm) {
  const seen=new Map();
  for(const stage of stages[arm])for(const row of stage.rows){
    if(!seen.has(row.path))seen.set(row.path,{stage:stage.name,rows:stage.rows.filter(r=>r.path===row.path)});
    // File repetition is checked once per second file occurrence, not per case.
  }
  if(arm==='E'){
    const first=stages.E.find(s=>s.name==='text-readers'),second=stages.E.find(s=>s.name==='runtime');
    const a=first.rows.filter(r=>r.path===FLAT),b=second.rows.filter(r=>r.path===FLAT);
    requireThat(a.length===56&&b.length===56,'Declared flat-layout repetition missing/changed');
    equal(a.map(r=>r.id).sort(),b.map(r=>r.id).sort(),'Repeated flat-layout identity mismatch');
    equal(a.map(r=>JSON.stringify([r.id,outcome(r.original)])).sort(),b.map(r=>JSON.stringify([r.id,outcome(r.original)])).sort(),'Repeated flat-layout outcome mismatch');
    requireThat(first.clean&&second.clean,'Both repeated flat-layout stage occurrences must independently pass');
  }
  // Only the declared repetition may be folded. Duplicate tuples within a file
  // are preserved; they are not removed by Set.
  const selections=(arm==='B'?schedule:eSchedule).stages.flatMap(s=>s.files);
  const duplicates=selections.filter((p,i)=>selections.indexOf(p)!==i);
  equal(duplicates,arm==='E'?[FLAT]:[],'Undeclared repeated file occurrence');
  const rows=[...seen.values()].flatMap(f=>f.rows);
  requireThat(rows.length===(arm==='B'?5510:6739),'Unique population incomplete');
  uniqueRows[arm]=rows;
}
function exactComparison() {
  const keys=d=>Object.entries(d).flatMap(([p,ids])=>ids.map(id=>caseKey(p,id))).sort();
  const authority=partition(keys(expectedB),keys(expectedE));
  const actual=partition(uniqueRows.B.map(r=>r.key),uniqueRows.E.map(r=>r.key));
  const split=part=>({
    shared:part.shared,
    BOnlyCommon:part.BOnly.filter(k=>JSON.parse(k)[0]!==PROJECTION),
    BProjection:part.BOnly.filter(k=>JSON.parse(k)[0]===PROJECTION),
    EOnlyCommon:part.EOnly.filter(k=>Object.hasOwn(expectedB,JSON.parse(k)[0])),
    EOnlyFiles:part.EOnly.filter(k=>!Object.hasOwn(expectedB,JSON.parse(k)[0]))
  });
  const wanted=split(authority),got=split(actual);
  equal(Object.values(wanted).map(a=>a.length),[5406,2,102,16,1317],'Frozen authority partition count mismatch');
  // Independently bind the literal common-file delta inventory in the B schedule.
  const declaredBO=schedule.comparison.rows.flatMap(r=>r.BOnly.map(id=>caseKey(r.path,id))).sort();
  const declaredEO=schedule.comparison.rows.flatMap(r=>r.EOnly.map(id=>caseKey(r.path,id))).sort();
  equal(wanted.BOnlyCommon,declaredBO,'B-only exact partition authority mismatch');
  equal(wanted.EOnlyCommon,declaredEO,'E-only common exact partition authority mismatch');
  equal([...new Set(wanted.EOnlyFiles.map(k=>JSON.parse(k)[0]))].sort(),schedule.absentCandidateOnlyFiles,'E-only file partition mismatch');
  equal(wanted.BProjection,expectedB[PROJECTION].map(id=>caseKey(PROJECTION,id)).sort(),'102-control exact partition mismatch');
  report.partition={expected:wanted,observed:got,counts:Object.fromEntries(Object.entries(got).map(([k,v])=>[k,v.length]))};
  equal(got,wanted,'Observed exact partition differs from frozen authority');
  report.identityEquality.exactPartition=true;
  const sharedKeys=new Set(wanted.shared),wantedCounts=counts(wanted.shared);
  const select=rows=>rows.filter(r=>sharedKeys.has(r.key));
  const b=select(uniqueRows.B),e=select(uniqueRows.E);
  // All shared case copies are comparable only when multiplicities agree.
  equal(expand(counts(b.map(r=>r.key))),expand(wantedCounts),'B shared multiset mismatch');
  equal(expand(counts(e.map(r=>r.key))),expand(wantedCounts),'E shared multiset mismatch');
  report.identityEquality.sharedMultiset=true;
  const results=rows=>{
    const m=new Map();
    for(const r of rows){if(!m.has(r.key))m.set(r.key,[]);m.get(r.key).push(outcome(r.original));}
    for(const v of m.values())v.sort();return m;
  };
  const bm=results(b),em=results(e);
  for(const key of [...sharedKeys].sort()){
    if(JSON.stringify(bm.get(key))!==JSON.stringify(em.get(key)))report.outcomeEquality.differences.push({key,B:bm.get(key),E:em.get(key)});
  }
  report.outcomeEquality.shared=report.outcomeEquality.differences.length===0;
}
function main(){
  requireThat(fileURLToPath(import.meta.url)===SELF,'Wrong comparator path');
  requireThat(process.argv.length===6&&process.argv[2]==='--self-sha256'&&process.argv[4]==='--b-terminal-sha256','Exact explicit self/B terminal hash arguments required');
  const selfHash=process.argv[3],parentHash=process.argv[5];
  requireThat(digestRE.test(selfHash)&&digestRE.test(parentHash),'Full explicit SHA256 required; no trust-on-first-use');
  bytes(SELF,{sha256:selfHash});report.authentication.comparator=true;
  report.provenance.B.parentSuppliedTerminalSHA256=parentHash;
  const authorityStart=report.errors.length;attempt('authority',loadAuthorities);
  report.authentication.authorities=!!expectedB&&!!expectedE&&cleanSince(authorityStart,'authority');
  // Even authority failure must retain original B terminal/inventory where safe.
  attempt('B.authentication',()=>retainBInventory(parentHash));
  if(!schedule||!eSchedule||!baselines||!expectedB||!expectedE)return;
  const eStart=report.errors.length;attempt('E.authentication',loadEArchive);
  report.authentication.B=!!bTerminal&&!report.errors.some(e=>e.phase==='B.authentication');
  report.authentication.E=!!eTerminal&&cleanSince(eStart,'E.authentication');
  readStages('B',schedule.stages,schedule.roots.B,B_RUN,expectedB);
  readStages('E',eSchedule.stages,E_ROOT,E_RUN,expectedE);
  // Native evidence is retained for BOTH arms before any pass-first evaluation.
  evaluateStages('B',schedule.stages,schedule.roots.B,B_RUN,B_RUNNER_SHA);
  evaluateStages('E',eSchedule.stages,E_ROOT,E_RUN,schedule.mechanismReference.sha256);
  rootGates();
  // Stage inventories can add authentication failures after initial root checks.
  report.authentication.B=!!bTerminal&&!report.errors.some(e=>e.phase==='B.authentication');
  report.authentication.E=!!eTerminal&&!report.errors.some(e=>e.phase==='E.authentication');
  for(const arm of ['B','E']){
    const allPresent=stages[arm].length===(arm==='B'?9:10)&&stages[arm].every(s=>s.available);
    const terminalPresent=arm==='B'?!!bTerminal:!!eTerminal;
    report.completeness[arm]=allPresent&&terminalPresent?'complete':'incomplete';
    const trusted=report.authentication[arm]&&report.authentication.authorities;
    report.identityEquality[arm+'ToAuthority']=trusted&&allPresent?stages[arm].every(s=>s.identityExact):null;
    report.evidenceCleanliness[arm]=trusted&&allPresent&&stages[arm].every(s=>s.clean)&&!report.errors.some(e=>e.phase===arm+'.cleanliness');
  }
  attempt('B.identity',()=>foldUnique('B'));
  // Use populated rows as the explicit folding completion indicator.
  attempt('E.identity',()=>foldUnique('E'));
  if(report.authentication.B&&report.authentication.E&&report.authentication.authorities&&report.completeness.B==='complete'&&report.completeness.E==='complete'&&uniqueRows.B.length===5510&&uniqueRows.E.length===6739){
    const start=report.errors.length;attempt('comparison.identity',exactComparison);
    if(!cleanSince(start,'comparison.identity')){
      report.identityEquality.exactPartition=false;
      if(report.identityEquality.sharedMultiset!==true)report.identityEquality.sharedMultiset=false;
      report.outcomeEquality.shared=null;
    }
  }
  report.evidenceCleanliness.all=report.evidenceCleanliness.B===true&&report.evidenceCleanliness.E===true;
}
attempt('preflight',main);
if(report.errors.some(e=>e.phase==='preflight'))report.evidenceCleanliness.all=false;
// Remove internal Maps from the JSON projection; original native bytes and
// assertion objects, including all failures, remain in retained evidence/rows.
for(const arm of ['B','E'])for(const s of report.stageEvidence[arm])delete s.indexed;
console.log(JSON.stringify(report,null,2));
if(report.evidenceCleanliness.all!==true||report.identityEquality.exactPartition!==true||report.identityEquality.sharedMultiset!==true||report.outcomeEquality.shared!==true)process.exitCode=1;
