function safe(p) {
  assert(typeof p === 'string' && p.length > 0 && !p.includes('\\') && !p.includes('\0'), 'unsafe path');
  assert(p.split('/').every(s => /^[A-Za-z0-9_.-]+$/.test(s) && s !== '.' && s !== '..'), p);
  assert.equal(relative(ROOT, resolve(ROOT, p)), p, p);
  return p;
}
function checked(p, directory = false) {
  safe(p);
  let current = ROOT;
  // Reject symlinks in the trusted root too, not just the relative leaf.
  let prefix = '/';
  for (const component of ROOT.split('/').filter(Boolean)) {
    prefix = join(prefix, component);
    const s = fs.lstatSync(prefix);
    assert(!s.isSymbolicLink() && s.isDirectory(), prefix);
  }
  const parts = p.split('/');
  for (let i = 0; i < parts.length; i++) {
    current = join(current, parts[i]);
    const s = fs.lstatSync(current);
    assert(!s.isSymbolicLink(), current);
    assert(i < parts.length - 1 || directory ? s.isDirectory() : s.isFile(), current);
  }
  return current;
}
function raw(p) {
  const filename = checked(p);
  const fd = fs.openSync(filename, fs.constants.O_RDONLY | fs.constants.O_NOFOLLOW);
  try {
    const before = fs.fstatSync(fd, {bigint: true});
    assert(before.isFile(), p);
    const bytes = fs.readFileSync(fd);
    const after = fs.fstatSync(fd, {bigint: true});
    const leaf = fs.lstatSync(filename, {bigint: true});
    for (const key of ['dev', 'ino', 'size', 'mtimeNs', 'ctimeNs']) {
      assert.equal(before[key], after[key], `changing input ${p}`);
      assert.equal(after[key], leaf[key], `replaced input ${p}`);
    }
    assert.equal(BigInt(bytes.length), after.size);
    return bytes;
  } finally { fs.closeSync(fd); }
}
function capture(p, expected) {
  if (!captured.has(p)) captured.set(p, raw(p));
  const bytes = captured.get(p);
  if (expected) assert.equal(sha(bytes), expected, p);
  return bytes;
}

function inventory(dir) {
  const names = fs.readdirSync(checked(dir, true)).sort();

  if (inventories.has(dir)) assert.deepEqual(names, inventories.get(dir), `inventory changed: ${dir}`);
  else inventories.set(dir, names);
  return names;
}

  return {dir: run.child, focusedRun: run.dir, root: ROOT, terminal, verdict,
    historicalPairExecuted: false, physicalAcceptanceCertified: false, accepted: false};
