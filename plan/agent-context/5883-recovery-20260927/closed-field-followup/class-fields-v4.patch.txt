+// Copyright (c) 2026 Loopdive GmbH. Licensed under Apache-2.0 WITH LLVM-exception.
+import type { FieldDef, Instr } from "../ir/types.js";
+import type { CodegenContext } from "./context/types.js";
+
+/** Read-only: safe in allocation assertion planners. No registration/fallback. */
+export function classReferenceDefaultInstrs(ctx: Pick<CodegenContext, "undefinedGlobalIdx">, field: FieldDef): Instr[] {
+  if (field.type.kind !== "externref") throw new Error("Expected class reference field");
+  if (!field.undefinedDefault) return [{ op: "ref.null.extern" }];
+  if (ctx.undefinedGlobalIdx === undefined) throw new Error("Class undefined default was not prepared");
+  return [{ op: "global.get", index: ctx.undefinedGlobalIdx }, { op: "extern.convert_any" }];
+}

+  const ctx = Object.freeze({ undefinedGlobalIdx: 0 });

+});

+  const b = classReferenceDefaultInstrs({ undefinedGlobalIdx: 7 }, marked);
+  expect(a).toEqual(b);
+  expect(a).not.toBe(b);
+  expect(a[0]).not.toBe(b[0]);
