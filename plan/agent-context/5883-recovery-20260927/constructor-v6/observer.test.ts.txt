const sha = (value: string | Uint8Array) => createHash("sha256").update(value).digest("hex");
function encode(root: unknown): string {
  const seen = new Map<object, number>();
  function visit(value: unknown): unknown {
    if (value === undefined) return { atom: "undefined" };
    if (typeof value === "bigint") return { bigint: String(value) };
    if (typeof value === "number" && (!Number.isFinite(value) || Object.is(value, -0)))
      return { number: Object.is(value, -0) ? "-0" : String(value) };
    if (typeof value === "function" || typeof value === "symbol")
      throw new Error(`Unsupported snapshot value: ${typeof value}`);
    if (value === null || typeof value !== "object") return value;
    const previous = seen.get(value);
    if (previous !== undefined) return { ref: previous };
    const id = seen.size;
    seen.set(value, id);
    if (value instanceof Map) return { id, map: [...value].map(([k, v]) => [visit(k), visit(v)]) };
    if (value instanceof Set) return { id, set: [...value].map(visit) };
    if (ArrayBuffer.isView(value))
      return { id, view: value.constructor.name, bytes: [...new Uint8Array(value.buffer, value.byteOffset, value.byteLength)] };
    if (value instanceof ArrayBuffer) return { id, buffer: [...new Uint8Array(value)] };
    const entries = Reflect.ownKeys(value).map((key) => {
      if (typeof key !== "string") throw new Error("Symbol snapshot key");
      const descriptor = Object.getOwnPropertyDescriptor(value, key)!;
      if (!("value" in descriptor)) throw new Error(`Accessor snapshot key: ${key}`);
      return [key, visit(descriptor.value)];
    });
    return { id, kind: Array.isArray(value) ? "array" : "object", entries };
  }
  return JSON.stringify(visit(root));
}


// Read-only identity reachability; no providers, lazy accessors or getters.
function reachableObjects(roots: readonly unknown[]): Set<object> {
  const seen = new Set<object>();
  const visit = (value: unknown): void => {
    if (value === null || typeof value !== "object" || seen.has(value)) return;
    seen.add(value);
    if (value instanceof Map) { for (const [key, item] of value) { visit(key); visit(item); } return; }
    if (value instanceof Set) { for (const item of value) visit(item); return; }
    for (const descriptor of Object.values(Object.getOwnPropertyDescriptors(value))) {
      if (!("value" in descriptor)) throw new Error("Accessor in physical witness graph");
      visit(descriptor.value);
    }
  };
  // Only caller-supplied emitter-owned function bodies are admissible roots.
  // A held FunctionContext or module metadata is NEVER a reachability root.
  roots.forEach(visit);
  return seen;
}

function sourceFor(row: Case): string {
  if (row.group === "seam") return `export function test(marker: any): number {
  ${row.state === "already-emitted" ? "const warm: any = Promise;" : ""}
  const first: any = Promise;
  const second: any = Promise;
  const proto: any = Promise.prototype;
  const key: any = "constructor";
  const descriptor: any = Object.getOwnPropertyDescriptor(proto, key);
  return ${row.state === "already-emitted" ? "warm === first && " : ""}first === second && descriptor.value === first ? 1 : -1;
}`;
  return `declare function __drain_microtasks(): void;
export function test(marker: any): number {
  let answer = -100;
  const before: any = Promise;
  const input: any[] = [1, 2];
  (Promise as any).resolve = function(value: any): any {
    return {then: function(ok: any): void { ok(value); }};
  };
  const aggregate = Promise.${row.method}(input).then(function(value: any): void {
    answer = ${row.method === "all" ? "value[0] * 10 + value[1]" : "value"};
  }, function(): void { answer = -200; });
  const after: any = Promise;
  const proto: any = Promise.prototype;
  const key: any = "constructor";
  const descriptor: any = Object.getOwnPropertyDescriptor(proto, key);
  __drain_microtasks();
  return before === after && descriptor.value === before && answer === ${row.method === "all" ? 12 : 1} ? 1 : -1;
}`;
}

function existingSlot(ctx: CodegenContext, name: string): unknown {
  const descriptor = Object.getOwnPropertyDescriptor(ctx, name);
  if (!descriptor) return undefined;
  if (!("value" in descriptor)) throw new Error(`Accessor observation slot: ${name}`);
  return descriptor.value;
}
function physical(ctx: CodegenContext, fctx: FunctionContext): string {
  return encode({
    module: ctx.mod, funcMap: ctx.funcMap, structMap: ctx.structMap,
    structFields: ctx.structFields, typeNames: ctx.typeIdxToStructName,
    stringGlobals: ctx.stringGlobalMap, nativeStringGlobals: ctx.nativeStrLiteralGlobals,
    nativeStringHelpers: ctx.nativeStrHelpers, ctorGlobals: ctx.builtinObjectGlobals,
    imports: [ctx.numImportFuncs, ctx.numImportGlobals], errors: ctx.errors,
    pendingShift: ctx.pendingLateImportShift, liveBodies: ctx.liveBodies,
    caller: { name: fctx.name, params: fctx.params, locals: fctx.locals,
      localMap: fctx.localMap, body: fctx.body, savedBodies: fctx.savedBodies,
      returnType: fctx.returnType, blockDepth: fctx.blockDepth,
      breakStack: fctx.breakStack, continueStack: fctx.continueStack, labelMap: fctx.labelMap },
    sourceDerivedFlags: { standalone: ctx.standalone, nativeStrings: ctx.nativeStrings,
      protoNamedDirty: ctx.protoNamedDirty, protoMemberDirty: ctx.protoMemberDirty,
      protoNamedWrittenMembers: ctx.protoNamedWrittenMembers },
    seeders: existingSlot(ctx, "__nativeProtoSeeders"),
    seededCtors: existingSlot(ctx, "__nativeProtoSeededCtors"),
    pendingSeeds: existingSlot(ctx, "__nativeProtoPendingSeeds"),
  });
}

        imports = WebAssembly.Module.imports(module);
        const instance = await WebAssembly.instantiate(module, {});
        actual = (instance.exports.test as (marker: unknown) => number)(null);
      } catch (error) { runtimeError = String(error); }
    }
    // Evaluate evidence only after the real observed transactions settle.
    // A callback event alone is never proof its enclosing emission survived.
    const returnedBinaryHash = result ? sha(result.binary) : null;
    const matchingEmissions = finalEmissions.filter((entry) => entry.completed && entry.binaryHash === returnedBinaryHash);
    const finalEmission = matchingEmissions.length === 1 ? matchingEmissions[0] : undefined;
    const evaluateWitness = (witness: Witness, finalEmission: FinalEmission | undefined) => {
      const seed = witness.seed, boundary = seed.boundary;
      const transactionIds = [...new Set([...seed.transactionIds, ...witness.transactionIds])];
      const ancestorOutcomes = transactionIds.map((id) => ({ id, status: transactions[id]!.status }));
      const ownOutcome = witness.ownTransactionId === undefined ? null : transactions[witness.ownTransactionId]!.status;
      let publicationReachable = false, anchorsReachable = false;
      const callerFunctionOrdinal = finalEmission && boundary?.bodyEmission
        ? finalEmission.functions.findIndex((entry) => entry.func === boundary.bodyEmission!.func) : -1;
      const moduleAttachmentKnown = finalEmission !== undefined && boundary !== undefined &&
        finalEmission.mod === boundary.ctx.mod && boundary.handoffVerified &&
        boundary.bodyEmission?.completed === true && callerFunctionOrdinal >= 0;
      record(() => {
        if (!moduleAttachmentKnown || !finalEmission) return;
        const reachable = finalEmission.reachable;
        publicationReachable = seed.publishInstruction !== undefined && reachable.has(seed.publishInstruction);
        anchorsReachable = witness.anchors.length > 0 && witness.anchors.every((instruction) => reachable.has(instruction));
      });
      const scopeReasons: string[] = [];
      if (!seed.eligible || seed.name !== "Promise") scopeReasons.push("not a Promise seed descended from the selected real frame");
      if (!boundary?.completed) scopeReasons.push("selected boundary did not complete");
      if (!moduleAttachmentKnown) scopeReasons.push("UNKNOWN: no exact producer-to-final-module caller attachment");
      if (!seed.completed) scopeReasons.push("seed did not complete");
      if (!seed.publication) scopeReasons.push("actual publication prefix not validated");
      if (!publicationReachable) scopeReasons.push("publication instruction not reachable in surviving physical output");
      if (ancestorOutcomes.some((entry) => entry.status !== "committed")) scopeReasons.push("enclosing transaction not committed");
      const sourceScopeSurvives = scopeReasons.length === 0;
      const emittedAnchorsSurvive = sourceScopeSurvives && anchorsReachable;
      type Proposition =
        | { kind: "seed"; publicationObserved: boolean; outputSurvival: "NOT_APPLICABLE" }
        | { kind: "commit"; callbackCommitted: boolean; committedOutputSurvives: boolean }
        | { kind: "compiler-reentry"; publishedReadSurvives: boolean; compilerOrderOnly: true }
        | { kind: "decline"; callbackDeclined: boolean; scopedDeclineObserved: boolean;
            outputSurvival: "NOT_APPLICABLE"; rollbackVerified: "UNOBSERVED" };
      let proposition: Proposition;
      switch (witness.kind) {
        case "seed":
          proposition = { kind: "seed", publicationObserved: sourceScopeSurvives, outputSurvival: "NOT_APPLICABLE" };
          break;
        case "commit":
          proposition = { kind: "commit", callbackCommitted: ownOutcome === "committed",
            committedOutputSurvives: ownOutcome === "committed" && emittedAnchorsSurvive };
          break;
        case "compiler-reentry":
          proposition = { kind: "compiler-reentry", publishedReadSurvives: emittedAnchorsSurvive && witness.samePublishedGlobal === true,
            compilerOrderOnly: true };
          break;
        case "decline":
          proposition = { kind: "decline", callbackDeclined: ownOutcome === "declined",
            scopedDeclineObserved: sourceScopeSurvives && ownOutcome === "declined",
            outputSurvival: "NOT_APPLICABLE", rollbackVerified: "UNOBSERVED" };
      }
      return { kind: witness.kind, seedId: seed.id, seedName: seed.name,
        seedFrame: seed.fctx.name, parentSeedIds: seed.parentSeedIds,
        boundaryId: boundary?.id ?? null, boundaryLabel: boundary?.label ?? null,
        ownTransactionId: witness.ownTransactionId ?? null, ownOutcome, ancestorOutcomes,
        moduleAttachment: { status: moduleAttachmentKnown ? "ATTACHED" : "UNKNOWN",
          emissionId: finalEmission?.id ?? null, bodyEmissionId: boundary?.bodyEmission?.id ?? null,
          callerFunctionOrdinal, handoffVerified: boundary?.handoffVerified ?? false },
        publicationPrefixValidated: seed.publication, sourceScopeSurvives, scopeReasons,
        emittedAnchorsSurvive, publicationReachable, anchorsReachable, anchorCount: witness.anchors.length,
        samePublishedGlobal: witness.samePublishedGlobal ?? null, proposition };

    };
    const witnessRecords = witnesses.map((witness) => evaluateWitness(witness, finalEmission));
    const outputPositive = (entry: ReturnType<typeof evaluateWitness>) =>
      entry.proposition.kind === "commit" ? entry.proposition.committedOutputSurvives :
      entry.proposition.kind === "compiler-reentry" ? entry.proposition.publishedReadSurvives : false;
    // Diagnostic shadow views only: never mutate the compiler module, functions,
    // fctx, instructions or binary. Every outcome uses the SAME actual predicate.
    const contractFailures: string[] = [];
    const contractRows: Array<{
      label: string; witnessIndex: number; expectedOutputSurvival: boolean;
      actual: ReturnType<typeof evaluateWitness>; passed: boolean;
      oldHeldBodyReachable?: boolean;
    }> = [];
    const positives = witnessRecords.flatMap((entry, index) => outputPositive(entry) && entry.anchorCount > 0 ? [index] : []);
    const assertView = (label: string, index: number, witness: Witness,
      view: FinalEmission | undefined, expectedOutputSurvival: boolean, oldHeldBodyReachable?: boolean) => {
      const actual = evaluateWitness(witness, view);
      const passed = outputPositive(actual) === expectedOutputSurvival;
      contractRows.push({ label, witnessIndex: index, expectedOutputSurvival, actual, passed, oldHeldBodyReachable });
      if (!passed) contractFailures.push(label);
      return actual;
    };
    record(() => {
      if (!finalEmission || positives.length === 0) {
        contractFailures.push("INCONCLUSIVE: no real nonempty eligible positive witness");
        return;
      }
      const index = positives.find((candidate) => {
        const witness = witnesses[candidate]!;
        const held = reachableObjects([witness.seed.boundary!.fctx.body]);
        return witness.seed.publishInstruction !== undefined && held.has(witness.seed.publishInstruction) &&
          witness.anchors.every((anchor) => held.has(anchor));
      });
      if (index === undefined) {
        contractFailures.push("INCONCLUSIVE: no real positive also reachable from held caller body alone");
        return;
      }
      const witness = witnesses[index]!;
      const boundary = witness.seed.boundary!;
      const caller = boundary.bodyEmission!.func;
      const shadow = (functions: FinalEmission["functions"]): FinalEmission => ({
        ...finalEmission, functions,
        reachable: reachableObjects(functions.map((entry) => entry.body)),
      });
      // Held body ALONE must witness the old false acceptance; the unchanged
      // real module cannot accidentally provide this negative's positive floor.
      const held = reachableObjects([boundary.fctx.body]);
      const oldHeldBodyReachable = witness.anchors.every((anchor) => held.has(anchor)) &&
        witness.seed.publishInstruction !== undefined && held.has(witness.seed.publishInstruction);
      if (!oldHeldBodyReachable) contractFailures.push("old held-body positive control did not fire");
      assertView("real-nonempty-positive", index, witness, finalEmission, true, oldHeldBodyReachable);
      assertView("orphan-caller-held-body-retained", index, witness,
        shadow(finalEmission.functions.filter((entry) => entry.func !== caller)), false, oldHeldBodyReachable);
      assertView("same-name-distinct-allocation", index, witness,
        shadow(finalEmission.functions.map((entry) => entry.func === caller
          ? { func: { ...entry.func }, body: entry.body } : entry)), false, oldHeldBodyReachable);
      // Remove all emitted roots, retaining all actual held references. This is
      // a diagnostic adverse view, NOT a claim real compilation emitted empties.
      assertView("all-bodies-detached-held-metadata-retained", index, witness,
        shadow(finalEmission.functions.map((entry) => ({ ...entry, body: [] }))), false, oldHeldBodyReachable);
      assertView("missing-final-emission", index, witness, undefined, false);
      assertView("nonempty-positive-repeated", index, witness, finalEmission, true);
      const committedIndex = witnessRecords.findIndex((entry) =>
        entry.proposition.kind === "commit" && entry.proposition.committedOutputSurvives && entry.anchorCount > 0);
      if (committedIndex < 0) {
        contractFailures.push("INCONCLUSIVE: no real nonempty surviving committed callback for empty-output control");
      } else {
        const empty = assertView("actual-committed-callback-empty-output", committedIndex,
          { ...witnesses[committedIndex]!, anchors: [] }, finalEmission, false);
        if (empty.proposition.kind !== "commit" || !empty.proposition.callbackCommitted || empty.proposition.committedOutputSurvives || empty.anchorsReachable)
          contractFailures.push("empty callback output conflated with survival");
      }
    });
    // Publication and decline controls never masquerade as output positives.
    const kindControlFailures: string[] = [];
    const kindControls: Array<{ label: string; witnessIndex: number;
      status: "OBSERVED" | "INCONCLUSIVE"; actual: ReturnType<typeof evaluateWitness> | null }> = [];
    record(() => {
      const seedIndex = witnessRecords.findIndex((entry) => entry.proposition.kind === "seed" &&
        entry.proposition.publicationObserved && entry.anchorCount === 0);
      if (seedIndex < 0 || !finalEmission) {
        kindControls.push({ label: "real-empty-anchor-seed-publication", witnessIndex: -1, status: "INCONCLUSIVE", actual: null });
        kindControls.push({ label: "orphan-seed-publication", witnessIndex: -1, status: "INCONCLUSIVE", actual: null });
        kindControlFailures.push("no real empty-anchor seed publication positive");
      } else {
        const seedWitness = witnesses[seedIndex]!;
        const actual = evaluateWitness(seedWitness, finalEmission);
        kindControls.push({ label: "real-empty-anchor-seed-publication", witnessIndex: seedIndex, status: "OBSERVED", actual });
        const caller = seedWitness.seed.boundary!.bodyEmission!.func;
        const functions = finalEmission.functions.filter((entry) => entry.func !== caller);
        const orphan = evaluateWitness(seedWitness, { ...finalEmission, functions,
          reachable: reachableObjects(functions.map((entry) => entry.body)) });
        kindControls.push({ label: "orphan-seed-publication", witnessIndex: seedIndex, status: "OBSERVED", actual: orphan });
        if (actual.emittedAnchorsSurvive || actual.proposition.kind !== "seed" || !actual.proposition.publicationObserved ||
          orphan.proposition.kind !== "seed" || orphan.proposition.publicationObserved)
          kindControlFailures.push("seed publication/output discrimination failed");
      }
      for (const [label, predicate] of [
        ["real-scoped-decline", (entry: ReturnType<typeof evaluateWitness>) =>
          entry.proposition.kind === "decline" && entry.proposition.scopedDeclineObserved],
        ["real-decline-with-rolled-back-ancestor", (entry: ReturnType<typeof evaluateWitness>) =>
          entry.proposition.kind === "decline" && entry.proposition.callbackDeclined &&
          // The actual predicate must reject ONLY for the ancestor outcome.
          // No altered ancestry or unrelated/unattached decline may supply this control.
          entry.scopeReasons.length === 1 && entry.scopeReasons[0] === "enclosing transaction not committed" &&
          entry.ancestorOutcomes.every((ancestor) =>
            ancestor.status === "committed" || ancestor.status === "declined" || ancestor.status === "rolled-back") &&
          entry.ancestorOutcomes.some((ancestor) => ancestor.status === "declined" || ancestor.status === "rolled-back")],
      ] as const) {
        const index = witnessRecords.findIndex(predicate);
        if (index < 0) {
          kindControls.push({ label, witnessIndex: -1, status: "INCONCLUSIVE", actual: null });
          kindControlFailures.push(label + ": no real capture");
          continue;
        }
        const actual = evaluateWitness(witnesses[index]!, finalEmission);
        kindControls.push({ label, witnessIndex: index, status: "OBSERVED", actual });
        if (actual.proposition.kind !== "decline" || actual.proposition.rollbackVerified !== "UNOBSERVED" ||
          (label === "real-decline-with-rolled-back-ancestor" && actual.proposition.scopedDeclineObserved))
          kindControlFailures.push(label + ": outcome incorrectly credited");
      }
    });
    const observerContract = { evidence: "diagnostic shadow views of real captured data",
      positiveCount: positives.length, rows: contractRows, kindControls,
      failures: [...contractFailures, ...kindControlFailures] };
    save("observer-contract.json", JSON.stringify(observerContract, null, 2));
    const scopedEvidence = {
      seedPublications: witnessRecords.filter((entry) => entry.proposition.kind === "seed" && entry.proposition.publicationObserved).length,
      committedOutputs: witnessRecords.filter((entry) => entry.proposition.kind === "commit" && entry.proposition.committedOutputSurvives).length,
      publishedCompilerReads: witnessRecords.filter((entry) => entry.proposition.kind === "compiler-reentry" && entry.proposition.publishedReadSurvives).length,
      scopedDeclineEvents: witnessRecords.filter((entry) => entry.proposition.kind === "decline" && entry.proposition.scopedDeclineObserved).length,
      verifiedDeclineRollbacks: 0,
    };
    const rollbackProof = { status: "UNOBSERVED", reason: "callback outcome is not restoration proof" };
    const coverageFailures: string[] = [];
    for (const [name, count] of Object.entries(scopedEvidence))
      if (count === 0) coverageFailures.push(name + " not observed");
    coverageFailures.push("runtime seed reentry unobserved: binaries intentionally uninstrumented");
    const basis = Object.fromEntries([
      "src/codegen/promise-combinator-drive.ts", "src/codegen/promise-combinator-observable-protocol.ts",
      "src/codegen/builtin-static-globals.ts", "src/codegen/builtin-constructor-prepared.ts",
      "src/codegen/staging-bindings.ts", "src/codegen/builtin-ctor-own-props.ts",
      "src/codegen/builtin-proto-constructor-seed.ts", "src/codegen/context/speculative.ts",
      "src/codegen/native-proto.ts", "src/codegen/function-body.ts", "src/emit/binary.ts", "src/compiler.ts",
      "tests/issue-5883-real-constructor-observer-v6.test.ts",
    ].map((path) => [path, sha(readFileSync(resolve(path)))]));
    save("receipt.json", JSON.stringify({ arm, row, basis, sourceHash: sha(source),
      selected, sourceReads, readEvidence, driveHits, observerErrors, snapshots, probes,
      compilationWideDiagnostics: { seedCalls, committed, declined, reentries, publishedReentries },
      scopedEvidence, witnessRecords, transactions, observerContract, rollbackProof,
      finalModuleEvidence: { returnedBinaryHash, matchingEmissionIds: matchingEmissions.map((entry) => entry.id),
        emissions: finalEmissions.map((entry) => ({ id: entry.id, completed: entry.completed,
          binaryHash: entry.binaryHash, snapshot: entry.snapshot, functionCount: entry.functions.length })) },
      selectedBoundaries: boundaries.map(({ id, label, completed }) => ({ id, label, completed })),
      seedRecords: seedSpans.map((seed) => ({ id: seed.id, name: seed.name, frame: seed.fctx.name,
        boundaryId: seed.boundary?.id ?? null, parentSeedIds: seed.parentSeedIds,
        transactionIds: seed.transactionIds, completed: seed.completed, publication: seed.publication,
        boundaryContextMatches: seed.boundary?.ctx === seed.ctx,
        boundaryFrameMatches: seed.boundary?.fctx === seed.fctx })),
      transientEvidence: { files: roots, regionCalls, transfers },
      events, coverageFailures, runtimeReentry: { status: "UNOBSERVED", instrumentation: "none" },
      compileError, success: result?.success, errors: result?.errors,
      binaryHash: result ? sha(result.binary) : null, nativeActual, nativeError, actual, runtimeError, imports,
    }, null, 2));
    expect(observerErrors).toEqual([]);
    expect(observerContract.positiveCount).toBeGreaterThan(0);
    expect(observerContract.failures).toEqual([]);
    expect(observerContract.rows).toHaveLength(7);
    if (row.group === "seam") {
      expect(selected).toHaveLength(2);
      expect(driveHits).toHaveLength(0);
      expect(selected.every((hit) => hit.active === true)).toBe(true);
      if (row.state === "cold" || row.state === "controlled-reserved") expect(selected[0]!.beforeReservation).toBe(false);
      else expect(selected[0]!.beforeReservation).toBe(true);
      expect(sourceReads).toBe(row.state === "already-emitted" ? 3 : 2);
      expect(regionCalls).toBe(arm === "C" ? 2 : 0);
      expect(transfers).toBe(arm === "C" ? 2 : 0);
    } else {
      expect(driveHits).toEqual([{ method: row.method, arg: "input", observable: true, captured: true }]);
      expect(readEvidence.some((hit) => hit.phase === "before-drive")).toBe(true);
      expect(readEvidence.some((hit) => hit.phase === "after-drive")).toBe(true);
      expect(regionCalls).toBe(arm === "C" ? 1 : 0);
      expect(transfers).toBe(arm === "C" ? 1 : 0);
    }
    expect(compileError).toBeNull();
    expect(result?.success, JSON.stringify(result?.errors)).toBe(true);
    expect(nativeError).toBeNull(); expect(nativeActual).toBe(1);
    expect(imports).toEqual([]); expect(runtimeError).toBeNull(); expect(actual).toBe(1);
    // Honest coverage failure is separate from preserved correspondence data.
    // This cannot go green merely because identical uninstrumented rows pass.
    expect(coverageFailures, "mandatory constructor coverage remains unmet").toEqual([]);
  }, 120000);
}
