+const allocation: VectorDomainAllocationRole = {
+  allocationRole: "literal:ordinary", layoutBinding: "layout:externref",

+function setup(options: { imports?: boolean; element?: "externref" | "f64"; twoRoles?: boolean } = {}) {
+  const module = createEmptyModule();
+  const tx = new PhysicalModuleReservations(module);
+  if (options.imports) {
+    tx.reserveFunctionImport("preceding:function", "prior", "fn", { params: [], results: [] });
+    tx.reserveGlobalImport("preceding:global", "prior", "number", { kind: "i32" }, false);
+  }
+  const element = options.element ?? "externref";
+  const backing = tx.reserveType("backing:" + element,
+    { kind: "array", name: "test_backing", element: { kind: element }, mutable: true });
+  const carrier = tx.reserveType("layout:" + element, {
+    kind: "struct", name: "test_existing_carrier", fields: [
+      { name: "length", type: { kind: "i32" }, mutable: true },
+      { name: "data", type: { kind: "ref", typeIdx: backing.typeIndex }, mutable: true },
+    ],
+  });
+  const row: VectorDomainAllocationRole = {
+    ...allocation, layoutBinding: carrier.key, storageKind: element === "f64" ? 2 : 1,
+  };
+  const rows = options.twoRoles ? [row, { ...row, allocationRole: "call:pack", semanticKind: 4 as const }] : [row];
+  const plan = planVectorDomains("test:domain", rows);
+  const bindings = rows.map((value) => ({ allocationRole: value.allocationRole,
+    layoutBinding: value.layoutBinding, carrier, backing }));
+  return { module, tx, carrier, backing, plan, bindings };
+}
+
+function reserve(options: Parameters<typeof setup>[0] = {}) {
+  const value = setup(options);
+  return { ...value, pack: reserveVectorDomains(value.tx, value.plan, value.bindings) };
+}

+/**
+ * Test-only trapping bodies: exercise physical completion, NEVER semantic readiness.
+ * Every service and validator traps when called; no fake success implementation.
+ */
+function physicalOnlyFills(pack: ReservedVectorDomains): VectorDomainFunctionFill[] {
+  return pack.rows.flatMap((row) => [row.validator, ...row.operations.map((entry) => entry.function)])
+    .map((target): VectorDomainFunctionFill => ({ target, locals: [], body: [{ op: "unreachable" }] }));
+}

+function population(module: ReturnType<typeof createEmptyModule>) {
+  return {
+    types: module.types.length, functions: module.functions.length, globals: module.globals.length,
+    imports: module.imports.length, tables: module.tables.length, exports: module.exports.length,
+    tags: module.tags.length, memories: module.memories.length, elements: module.elements.length,
+    strings: module.stringPool.length, data: module.dataSegments.length,
+  };

+/** Emit-only diagnostic accessors; not production field2 attachment or public service publication. */
+function binaryFixture(options: { imports?: boolean; f64?: boolean; invalidResult?: boolean } = {}) {
+  const value = reserve({ imports: options.imports, element: options.f64 ? "f64" : "externref", twoRoles: true });
+  const roles = vectorDomainPhysicalRoles(value.tx, value.pack);
+  const row = value.pack.rows[0]!;
+  const read = (name: string, results: ValType[]) => value.tx.reserveFunction("test:" + name, name, { params: [], results });
+  const origin = read("origin", [{ kind: "externref" }]);
+  const domain = read("domain", [{ kind: "externref" }]);
+  const hole = read("hole", [{ kind: "externref" }]);
+  const info = read("info", [{ kind: "i32" }, { kind: "i32" }, { kind: "i32" }]);
+  const shared = read("sharedOrigin", [{ kind: "i32" }]);
+  const invoke = read("invokeRawLength", [{ kind: "i32" }]);
+  value.tx.freezeReservations();
+  const definitions = physicalOnlyFills(value.pack);
+  if (options.invalidResult) {
+    definitions.find((entry) => entry.target === row.operations[0]!.function)!.body = [{ op: "ref.null.extern" }];
+  }
+  const filled = fillVectorDomainServices(value.tx, value.pack, definitions);
+  const domainGet = (index = 0): Instr => ({ op: "global.get", index: value.tx.physicalIndex(value.pack.rows[index]!.domain) });
+  const field = (fieldIdx: number): Instr => ({ op: "struct.get", typeIdx: roles.domainType.typeIndex, fieldIdx });
+  value.tx.fillFunction(origin, { locals: [], body: [domainGet(), field(3), { op: "extern.convert_any" }] });
+  value.tx.fillFunction(domain, { locals: [], body: [domainGet(), { op: "extern.convert_any" }] });
+  value.tx.fillFunction(hole, { locals: [], body: [domainGet(), field(4)] });
+  value.tx.fillFunction(info, { locals: [], body: [domainGet(), field(0), domainGet(), field(1), domainGet(), field(2)] });
+  value.tx.fillFunction(shared, { locals: [], body: [domainGet(), field(3), domainGet(1), field(3), { op: "ref.eq" }] });
+  value.tx.fillFunction(invoke, { locals: [], body: [
+    { op: "ref.null.extern" }, domainGet(), field(5),
+    { op: "struct.get", typeIdx: roles.servicesType.typeIndex, fieldIdx: 0 },
+    { op: "call_ref", typeIdx: row.operations[0]!.function.object.typeIdx },
+  ] });
+  for (const fn of [origin, domain, hole, info, shared, invoke]) value.tx.defineExport("export:" + fn.key, fn.object.name, fn);
+  value.tx.seal();
+  sealVectorDomainResources(value.tx, filled);
+  return { bytes: Uint8Array.from(emitBinary(value.module)), value };
+}

+interface ProbeExports {
+  origin(): unknown;
+  domain(): unknown;
+  hole(): unknown;
+  info(): number[];
+  sharedOrigin(): number;
+  invokeRawLength(): number;
+}
+function probes(instance: WebAssembly.Instance): ProbeExports {
+  return instance.exports as unknown as ProbeExports;
+}
