+function context(entries: [string, number][] = []) {
+  // Deliberately incomplete: these boundaries require only the canonical map.
+  // Mutation sentinels cover module, local and registration state as well.
+  const state = {
+    funcMap: new Map(entries),
+    mod: { functions: [], types: [], imports: [], globals: [] },
+    locals: [],
+    registrationState: { phase: "prepared" },
+  };
+  return { state, ctx: state as unknown as CodegenContext };
+}

+for (const [name, provider, build] of [
+  ["truthiness", "__is_truthy", runtimeToBooleanInstrs],
+  ["number string", "number_toString", runtimeNumberToStringInstrs],
+] as const) {
+  it(name + ": missing provider is null without mutation", () => {
+    const { ctx, state } = context();
+    const before = structuredClone(state);
+    expect(build(ctx)).toBeNull();
+    expect(state).toEqual(before);
+  });
+  it(name + ": index zero is valid", () => {
+    const { ctx } = context([[provider, 0]]);
+    expect(build(ctx)).toEqual([{ op: "call", funcIdx: 0 }]);
+  });
+  it(name + ": reads a shifted current map rather than caching", () => {
+    const { ctx } = context([[provider, 7]]);
+    const first = build(ctx);
+    ctx.funcMap.set(provider, 19);
+    expect(build(ctx)).toEqual([{ op: "call", funcIdx: 19 }]);
+    expect(first).toEqual([{ op: "call", funcIdx: 7 }]);
+  });
+  it(name + ": instructions and arrays have fresh identity", () => {
+    const { ctx } = context([[provider, 2]]);
+    const a = build(ctx)!;
+    const b = build(ctx)!;
+    expect(a).not.toBe(b);
+    expect(a[0]).not.toBe(b[0]);
+    expect(a).toEqual(b);
+  });
+  it(name + ": present provider does not allocate or mutate", () => {
+    const { ctx, state } = context([[provider, 4]]);
+    const before = structuredClone(state);
+    const ensure = vi.spyOn(shared, "ensureLateImport").mockImplementation(() => {
+      throw new Error("unexpected preparation");
+    });
+    const union = vi.spyOn(index, "addUnionImports").mockImplementation(() => {
+      throw new Error("unexpected union warming");
+    });
+    expect(build(ctx)).toEqual([{ op: "call", funcIdx: 4 }]);
+    expect(state).toEqual(before);
+    expect(ensure).not.toHaveBeenCalled();
+    expect(union).not.toHaveBeenCalled();
+  });
