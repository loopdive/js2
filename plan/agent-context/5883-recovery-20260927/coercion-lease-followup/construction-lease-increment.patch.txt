+  const occurrence = graph.recordOccurrence(root, provenance("node"), { nodeIdentity, kind });
+  const node = stageLeaf({ op: "local.get", index: 0 }, { nodeIdentity, occurrence: { occurrence, kind } });
+  const staged = { body, ...identity, root: stageRegion(body, root, [node]) };
+  const manifest = inspectStageBody(staged).manifest;
+  const expected: StagePlacement[] = [{ region: root, parent: { kind: "body", body }, nodeIdentities: [nodeIdentity] }];
+  return {
+    graph,
+    identity,
+    body,
+    token,
+    root,
+    edge,
+    spare,
+    occurrence,
+    nodeIdentity,
+    node,
+    staged,
+    manifest,
+    expected,
+  };
+}
+type Candidate = ReturnType<typeof candidate>;
+const claim = (c: Candidate) => c.graph.claimBodyConstruction(c.body, c.token, c.identity);
+const editor = (c: Candidate) => new StagingBodyEditor(c.graph, c.staged, c.token, c.expected);
+function complete(c: Candidate, lease: ConstructionLease) {
+  return c.graph.completeConstruction(c.body, c.token, c.manifest, lease);
+}
+
+/** Publication sentinels observe the shared id allocator without changing body content. */
+function rejectsWithoutAllocation(graph: EmissionOwnership, action: () => unknown, message?: string) {
+  const before = graph.createPublication("function", provenance("before"));
+  if (message) expect(action).toThrow(message);
+  else expect(action).toThrow();
+  const after = graph.createPublication("function", provenance("after"));
+  expect(after.id).toBe(before.id + 1);
+}
+
+describe("arena-owned construction lease", () => {
+  it("rejects the wrong body's top token before committing or mutating either construction", () => {
+    const graph = new EmissionOwnership();
+    const a = candidate(graph),
+      b = candidate(graph);
+    rejectsWithoutAllocation(
+      graph,
+      () => new StagingBodyEditor(graph, a.staged, b.token, a.expected),
+      "construction owner",
+    );
+    const bl = claim(b);
+    graph.assertConstructionScope(b.body, b.token, bl);
+    complete(b, bl);
+    const al = claim(a);
+    graph.assertConstructionScope(a.body, a.token, al);
+    complete(a, al); // Both original manifests still authenticate exactly.
+  });
+
+  it("rejects cross-arena tokens and bodies without consuming either arena's scope", () => {
+    const a = candidate(),
+      b = candidate();
+    rejectsWithoutAllocation(a.graph, () => a.graph.claimBodyConstruction(a.body, b.token, a.identity), "LIFO");
+    rejectsWithoutAllocation(a.graph, () => a.graph.claimBodyConstruction(b.body, a.token, b.identity), "foreign");
+    complete(a, claim(a));
+    complete(b, claim(b));
+  });
+
+  it("a structural copy of the token cannot claim its original scope", () => {
+    const c = candidate();
+    rejectsWithoutAllocation(c.graph, () => c.graph.claimBodyConstruction(c.body, { ...c.token }, c.identity), "LIFO");
+    complete(c, claim(c));
+  });
+
+  it.each(["commit", "rollback"] as const)("rejects a stale token after unclaimed %s", (exit) => {
+    const c = candidate();
+    if (exit === "commit") c.graph.commitTransaction(c.token);
+    else c.graph.rollbackTransaction(c.token);
+    rejectsWithoutAllocation(c.graph, () => editor(c), "LIFO");
+    if (exit === "commit") c.graph.completeBodyEpoch(c.body, c.manifest);
+    else expect(() => c.graph.retainBodyEpoch(c.body)).toThrow("failed");
+  });
+
+  it.each(["caller", "unbound", "nested"] as const)("rejects a %s token as construction owner", (kind) => {
+    const c = candidate();
+    const wrong =
+      kind === "caller"
+        ? c.graph.beginTransaction()
+        : c.graph.beginContentTransaction(kind === "nested" ? c.body : undefined);
+    rejectsWithoutAllocation(
+      c.graph,
+      () => new StagingBodyEditor(c.graph, c.staged, wrong, c.expected),
+      "construction owner",
+    );
+    c.graph.rollbackTransaction(wrong);
+    complete(c, claim(c));
+  });
+
+  it.each(["bodyIdentity", "bindingEnvironment"] as const)("rejects stale %s before claiming", (field) => {
+    const c = candidate();
+    rejectsWithoutAllocation(
+      c.graph,
+      () => new StagingBodyEditor(c.graph, { ...c.staged, [field]: {} }, c.token, c.expected),
+      "Stale",
+    );
+    editor(c).complete();
+  });
+
+  it("rejects duplicate editor claims while the first editor remains usable", () => {
+    const c = candidate();
+    const first = editor(c);
+    const before = first.snapshot();
+    rejectsWithoutAllocation(c.graph, () => editor(c), "already claimed");
+    expect(first.snapshot()).toBe(before);
+    first.removeLeaf(c.nodeIdentity);
+    expect(inspectStageBody(first.complete().body).manifest.occurrences).toEqual([]);
+  });
+
+  it("a bad physical preflight does not consume the construction claim", () => {
+    const c = candidate();
+    rejectsWithoutAllocation(c.graph, () => new StagingBodyEditor(c.graph, c.staged, c.token, []), "region count");
+    editor(c).complete();
+  });
+
+  const rawWrites = [
+    "create-region",
+    "attach-region",
+    "record-occurrence",
+    "remove-occurrence",
+    "adopt-body",
+    "detach-edge",
+    "begin-content",
+    "commit",
+    "rollback",
+    "retain",
+    "complete",
+    "discard",
+  ] as const;
+  it.each(rawWrites)(
+    "rejects claimed raw %s before allocation, stack change, membership or state change",
+    (operation) => {
+      const c = candidate();
+      const lease = claim(c);
+      const membership = c.graph.occurrenceMembership(c.occurrence);
+      function act() {
+        switch (operation) {
+          case "create-region":
+            return c.graph.createRegion(c.body);
+          case "attach-region":
+            return c.graph.attachRegion(c.root, c.spare);
+          case "record-occurrence":
+            return c.graph.recordOccurrence(c.root, provenance("bad"), { nodeIdentity: {}, kind: "seed-edge" });
+          case "remove-occurrence":
+            return c.graph.removeOccurrence(c.occurrence);
+          case "adopt-body":
+            return c.graph.adoptBody(c.root, c.body, {});
+          case "detach-edge":
+            return c.graph.detachEdge(c.edge);
+          case "begin-content":
+            return c.graph.beginContentTransaction(c.body);
+          case "commit":
+            return c.graph.commitTransaction(c.token);
+          case "rollback":
+            return c.graph.rollbackTransaction(c.token);
+          case "retain":
+            return c.graph.retainBodyEpoch(c.body);
+          case "complete":
+            return c.graph.completeBodyEpoch(c.body, c.manifest);
+          case "discard":
+            return c.graph.discardBodyEpoch(c.body);
+        }
+      }
+      rejectsWithoutAllocation(c.graph, act, "requires its construction lease");
+      c.graph.assertConstructionScope(c.body, c.token, lease);
+      expect(c.graph.occurrenceMembership(c.occurrence)).toBe(membership);
+      const receipt = complete(c, lease);
+      const publication = c.graph.createPublication("function", provenance("root"));
+      c.graph.publishBody(publication, c.body);
+      const result = c.graph.sealWithManifest({ publications: [{ publication, receipt }], bodies: [c.manifest] });
+      expect(result.ownership.bodies).toEqual([c.body]);
+      expect(result.ownership.regions).toEqual([c.root]);
+      expect(result.ownership.occurrences.map((o) => o.occurrence)).toEqual([c.occurrence]);
+      expect(result.manifests[0]?.adoptions).toEqual([]);
+    },
+  );
+
+  it("raw binding of an unbound frame cannot enter a claimed body", () => {
+    const c = candidate(),
+      lease = claim(c);
+    const unbound = c.graph.beginContentTransaction();
+    rejectsWithoutAllocation(
+      c.graph,
+      () => c.graph.bindContentTransaction(unbound, c.body),
+      "requires its construction lease",
+    );
+    // Failed binding left the frame genuinely unbound, not a hidden child.
+    expect(() => c.graph.commitTransaction(unbound)).toThrow("unbound");
+    c.graph.rollbackTransaction(unbound);
+    c.graph.assertConstructionScope(c.body, c.token, lease);
+    complete(c, lease);
+  });
+
+  it("a raw caller-frame write is rejected without tainting a claimed construction", () => {
+    const c = candidate(),
+      lease = claim(c);
+    const caller = c.graph.beginTransaction();
+    rejectsWithoutAllocation(c.graph, () => c.graph.removeOccurrence(c.occurrence), "requires its construction lease");
+    c.graph.rollbackTransaction(caller);
+    complete(c, lease); // Sticky unscoped-content taint was not changed on refusal.
+  });
+
+  it("a copied or cross-arena lease is not authority", () => {
+    const a = candidate(),
+      b = candidate();
+    const al = claim(a),
+      bl = claim(b);
+    for (const wrong of [{ ...al }, bl]) {
+      rejectsWithoutAllocation(
+        a.graph,
+        () => a.graph.beginContentTransaction(a.body, wrong),
+        "requires its construction lease",
+      );
+      rejectsWithoutAllocation(
+        a.graph,
+        () => a.graph.removeOccurrence(a.occurrence, wrong),
+        "requires its construction lease",
+      );
+      rejectsWithoutAllocation(
+        a.graph,
+        () => a.graph.rollbackTransaction(a.token, wrong),
+        "requires its construction lease",
+      );
+    }
+    complete(a, al);
+    complete(b, bl);
+  });
+
+  it("a lease cannot operate on another claimed body or its top frame", () => {
+    const graph = new EmissionOwnership();
+    const a = candidate(graph),
+      al = claim(a);
+    const b = candidate(graph),
+      bl = claim(b);
+    rejectsWithoutAllocation(graph, () => graph.rollbackTransaction(b.token, al), "requires its construction lease");
+    rejectsWithoutAllocation(graph, () => graph.beginContentTransaction(a.body, al), "Wrong body transaction domain");
+    rejectsWithoutAllocation(
+      graph,
+      () => graph.completeConstruction(a.body, b.token, a.manifest, al),
+      "Wrong construction lease scope",
+    );
+    graph.assertConstructionScope(b.body, b.token, bl);
+    complete(b, bl);
+    complete(a, al);
+  });
+
+  it("a lease cannot be used against an unclaimed body or caller frame", () => {
+    const a = candidate(),
+      al = claim(a);
+    const b = candidate();
+    rejectsWithoutAllocation(b.graph, () => b.graph.createRegion(b.body, al), "foreign");
+    const caller = a.graph.beginTransaction();
+    rejectsWithoutAllocation(a.graph, () => a.graph.beginContentTransaction(a.body, al), "body-owned scope");
+    rejectsWithoutAllocation(a.graph, () => a.graph.removeOccurrence(a.occurrence, al), "body-owned scope");
+    rejectsWithoutAllocation(a.graph, () => a.graph.rollbackTransaction(caller, al), "unbound/caller");
+    a.graph.rollbackTransaction(caller);
+    complete(a, al);
+    b.graph.commitTransaction(b.token);
+    b.graph.completeBodyEpoch(b.body, b.manifest);
+  });
+
+  it("nested rollback retains the lease and restores exact membership", () => {
+    const c = candidate(),
+      lease = claim(c);
+    const membership = c.graph.occurrenceMembership(c.occurrence);
+    const child = c.graph.beginContentTransaction(c.body, lease);
+    c.graph.removeOccurrence(c.occurrence, lease);
+    rejectsWithoutAllocation(c.graph, () => c.graph.rollbackTransaction(child), "requires its construction lease");
+    c.graph.rollbackTransaction(child, lease);
+    c.graph.assertConstructionScope(c.body, c.token, lease);
+    expect(c.graph.occurrenceMembership(c.occurrence)).toBe(membership);
+    const later = c.graph.beginContentTransaction(c.body, lease);
+    c.graph.commitTransaction(later, lease);
+    complete(c, lease);
+  });
+
+  it("even the valid lease cannot commit a construction without authenticated completion", () => {
+    const c = candidate(),
+      lease = claim(c);
+    rejectsWithoutAllocation(c.graph, () => c.graph.commitTransaction(c.token, lease), "completeConstruction");
+    c.graph.assertConstructionScope(c.body, c.token, lease);
+    complete(c, lease);
+  });
+
+  it("authorized nested writes and discard roll back without revoking the construction lease", () => {
+    const c = candidate(),
+      lease = claim(c);
+    const child = c.graph.beginContentTransaction(c.body, lease);
+    const region = c.graph.createRegion(c.body, lease);
+    c.graph.attachRegion(c.root, region, lease);
+    c.graph.recordOccurrence(region, provenance("child"), { nodeIdentity: {}, kind: "seed-edge" }, lease);
+    c.graph.adoptBody(region, c.body, {}, lease);
+    c.graph.discardBodyEpoch(c.body, lease);
+    c.graph.rollbackTransaction(child, lease);
+    c.graph.assertConstructionScope(c.body, c.token, lease);
+    complete(c, lease); // Exact pre-child manifest; no child membership survived.
+  });
+
+  it.each(["identity", "occurrence", "body"] as const)("bad %s manifest cannot close the construction", (kind) => {
+    const c = candidate(),
+      lease = claim(c);
+    const other = candidate();
+    const bad =
+      kind === "identity"
+        ? { ...c.manifest, bindingEnvironment: {} }
+        : kind === "occurrence"
+          ? { ...c.manifest, occurrences: [] }
+          : other.manifest;
+    rejectsWithoutAllocation(c.graph, () => c.graph.completeConstruction(c.body, c.token, bad, lease));
+    c.graph.assertConstructionScope(c.body, c.token, lease);
+    complete(c, lease);
+    other.graph.rollbackTransaction(other.token);
+  });
+
+  it("completion and construction abort both revoke authority permanently", () => {
+    for (const terminal of ["complete", "abort"] as const) {
+      const c = candidate(),
+        lease = claim(c);
+      if (terminal === "complete") complete(c, lease);
+      else c.graph.rollbackTransaction(c.token, lease);
+      rejectsWithoutAllocation(c.graph, () => c.graph.createRegion(c.body, lease), "revoked");
+      rejectsWithoutAllocation(c.graph, () => c.graph.beginContentTransaction(c.body, lease), "revoked");
+      rejectsWithoutAllocation(c.graph, () => c.graph.discardBodyEpoch(c.body), "revoked");
+      rejectsWithoutAllocation(c.graph, () => c.graph.assertConstructionScope(c.body, c.token, lease), "revoked");
+    }
+  });
+
+  it("an editor's exact scope is checked before any physical or ledger edit", () => {
+    const c = candidate();
+    const owned = editor(c);
+    const before = owned.snapshot();
+    const caller = c.graph.beginTransaction();
+    rejectsWithoutAllocation(c.graph, () => owned.removeLeaf(c.nodeIdentity), "LIFO");
+    rejectsWithoutAllocation(c.graph, () => owned.beginEdit(), "LIFO");
+    expect(owned.snapshot()).toBe(before);
+    c.graph.rollbackTransaction(caller);
+    owned.removeLeaf(c.nodeIdentity);
+    owned.complete();
+  });
+
+  it("an exposed editor child token does not grant raw close authority", () => {
+    const c = candidate();
+    const owned = editor(c);
+    const before = owned.snapshot();
+    const child = owned.beginEdit();
+    owned.removeLeaf(c.nodeIdentity);
+    rejectsWithoutAllocation(c.graph, () => c.graph.commitTransaction(child), "requires its construction lease");
+    rejectsWithoutAllocation(c.graph, () => c.graph.rollbackTransaction(child), "requires its construction lease");
+    owned.rollbackEdit(child);
+    expect(owned.snapshot()).toBe(before);
+    expect(inspectStageBody(owned.complete().body).manifest.occurrences.map((o) => o.occurrence)).toEqual([
+      c.occurrence,
+    ]);
+  });
+
+  it("preserves ordinary unclaimed nested transaction and completion behavior", () => {
+    const c = candidate();
+    const child = c.graph.beginContentTransaction(c.body);
+    c.graph.removeOccurrence(c.occurrence);
+    c.graph.rollbackTransaction(child);
+    c.graph.commitTransaction(c.token);
+    c.graph.completeBodyEpoch(c.body, c.manifest);
+  });
+});
