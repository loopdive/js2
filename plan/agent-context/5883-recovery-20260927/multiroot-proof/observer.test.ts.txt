const sha = (value: string | Uint8Array) => createHash("sha256").update(value).digest("hex");

function encode(root: unknown): string {
  const seen = new Map<object, number>();
  function visit(value: unknown): unknown {
    if (value === undefined) return { atom: "undefined" };
    if (typeof value === "bigint") return { bigint: String(value) };
    if (typeof value === "number" && (!Number.isFinite(value) || Object.is(value, -0)))
      return { number: Object.is(value, -0) ? "-0" : String(value) };
    if (typeof value === "function" || typeof value === "symbol")
      throw new Error(`Unsupported snapshot value: ${typeof value}`);
    if (value === null || typeof value !== "object") return value;
    const previous = seen.get(value);
    if (previous !== undefined) return { ref: previous };
    const id = seen.size;
    seen.set(value, id);
    if (value instanceof Map) return { id, map: [...value].map(([k, v]) => [visit(k), visit(v)]) };
    if (value instanceof Set) return { id, set: [...value].map(visit) };
    if (ArrayBuffer.isView(value))
      return { id, view: value.constructor.name, bytes: [...new Uint8Array(value.buffer, value.byteOffset, value.byteLength)] };
    if (value instanceof ArrayBuffer) return { id, buffer: [...new Uint8Array(value)] };
    const entries = Reflect.ownKeys(value).map((key) => {
      if (typeof key !== "string") throw new Error("Symbol snapshot key");
      const descriptor = Object.getOwnPropertyDescriptor(value, key)!;
      if (!("value" in descriptor)) throw new Error(`Accessor snapshot key: ${key}`);
      return [key, visit(descriptor.value)];
    });
    return { id, kind: Array.isArray(value) ? "array" : "object", entries };
  }
  return JSON.stringify(visit(root));
}

function snapshot(ctx: CodegenContext, fctx: FunctionContext): string {
  return encode({
    module: ctx.mod,
    functions: ctx.funcMap,
    structs: ctx.structMap,
    fields: ctx.structFields,
    typeNames: ctx.typeIdxToStructName,
    stringGlobals: ctx.stringGlobalMap,
    nativeStringGlobals: ctx.nativeStrLiteralGlobals,
    nativeStringHelpers: ctx.nativeStrHelpers,
    numImportFuncs: ctx.numImportFuncs,
    numImportGlobals: ctx.numImportGlobals,
    // A selected physical caller view, NOT serialization of AST/checker state.
    caller: {
      name: fctx.name, params: fctx.params, locals: fctx.locals,
      localMap: fctx.localMap, body: fctx.body, savedBodies: fctx.savedBodies,
      returnType: fctx.returnType, blockDepth: fctx.blockDepth,
      breakStack: fctx.breakStack, continueStack: fctx.continueStack,
      labelMap: fctx.labelMap,
    },
    sourceDerivedFlags: {
      standalone: ctx.standalone, nativeStrings: ctx.nativeStrings,
      protoNamedDirty: ctx.protoNamedDirty, protoMemberDirty: ctx.protoMemberDirty,
      protoNamedWrittenMembers: ctx.protoNamedWrittenMembers,
    },
  });
}

function sourceFor(method: "all" | "race"): string {
  return `declare function __drain_microtasks(): void;
export function test(): number {
  let answer = -100;
  const input: any[] = [1, 2];
  const resolver: any = function(value: any): any {
    return {then: function(ok: any): void { ok(value); }};
  };
  const bound: any = Function.prototype.bind.apply(resolver, [undefined]);
  (Promise as any).resolve = bound;
  const aggregate = Promise.${method}(input).then(function(value: any): void {
    answer = ${method === "all" ? "value[0] * 10 + value[1]" : "value"};
  }, function(): void { answer = -200; });
  __drain_microtasks();
  return answer;
}`;
}

// Read only actual emitted arrays; preserve leaf identity before finalization.
// The two local indices identify resolve's predicate, excluding constructor
// and per-element then-callability checks. Unknown/missing shape fails below.
function inspectArms(body: unknown, anyLocal: number, callableLocal: number) {
  const seen = new Set<object>();
  const ids = new Map<object, number>();
  const identity = (value: object) => {
    if (!ids.has(value)) ids.set(value, ids.size);
    return ids.get(value)!;
  };
  const arms: Array<{ typeIdx: number; leafIds: number[]; leafValues: unknown[] }> = [];
  function visit(value: unknown): void {
    if (!value || typeof value !== "object" || seen.has(value)) return;
    seen.add(value);
    if (Array.isArray(value)) {
      for (let i = 0; i + 2 < value.length; i++) {
        const [get, test, branch] = value.slice(i, i + 3);
        if (get?.op !== "local.get" || get.index !== anyLocal || test?.op !== "ref.test" || branch?.op !== "if") continue;
        const leaves = branch.then;
        if (!Array.isArray(leaves) || leaves.length !== 2 || leaves[0]?.op !== "i32.const" || leaves[0].value !== 1 || leaves[1]?.op !== "local.set" || leaves[1].index !== callableLocal)
          throw new Error("Unrecognized actual resolve-callability arm");
        arms.push({ typeIdx: test.typeIdx, leafIds: leaves.map(identity), leafValues: leaves.map((leaf) => ({ ...leaf })) });
      }
      value.forEach(visit);
    } else {
      for (const descriptor of Object.values(Object.getOwnPropertyDescriptors(value))) {
        if (!("value" in descriptor)) throw new Error("Accessor in emitted instruction graph");
        visit(descriptor.value);
      }
    }
  }
  visit(body);
  return { arms, rootCount: arms.length, distinctRoots: new Set(arms.map((arm) => arm.typeIdx)).size, uniqueLeaves: ids.size };
}

  it(`real bound-resolve multi-root ${method}`, async () => {
    const output = process.env.MULTIROOT_OBSERVER_OUTPUT;
    const arm = process.env.MULTIROOT_OBSERVER_ARM;
    if (!output || !["A", "B", "C"].includes(arm ?? "")) throw new Error("Set fresh MULTIROOT_OBSERVER_OUTPUT and MULTIROOT_OBSERVER_ARM=A|B|C");
    mkdirSync(output, { recursive: true });
    const save = (suffix: string, value: string | Uint8Array) => writeFileSync(resolve(output, `${method}.${suffix}`), value, { flag: "wx" });
    const source = sourceFor(method);
    save("source.ts", source);
    const hits: Array<Record<string, unknown>> = [];

    const original = drive.emitStandalonePromiseCombinatorDrive;
    const spy = vi.spyOn(drive, "emitStandalonePromiseCombinatorDrive").mockImplementation((...args) => {
      const [ctx, fctx, actualMethod, arg, observable, options] = args;
      const beforeNames = new Set(fctx.localMap.keys());
      const hit: Record<string, unknown> = {
        method: actualMethod, arg: arg.getText(), observable,
        captured: options?.iteratorProtocol === "captured-source-vector",
        entryBoundTypeIdx: ctx.boundFnTypeIdx,
        entryBindProvider: ctx.funcMap.get("__bind_dyn") ?? null,
      };
      const ordinal = hits.length;
      hits.push(hit);
      const observe = (phase: string) => {
        try { save(`entry-${ordinal}-${phase}.json`, snapshot(ctx, fctx)); }
        catch (error) { observerErrors.push(`${phase}: ${String(error)}`); }
      };
      observe("before");
      try { return original(...args); }
      finally {
        observe("after");
        try {
          const added = [...fctx.localMap].filter(([name]) => !beforeNames.has(name));
          const any = added.filter(([name]) => name.startsWith("__comb_observable_callable_any"));
          const callable = added.filter(([name]) => /^__comb_observable_callable_\d/.test(name));
          hit.resolveLocals = { any, callable };
          if (any.length !== 1 || callable.length !== 1) throw new Error("Expected exactly one real resolve predicate local pair");

          hit.observation = inspectArms(fctx.body, any[0]![1], callable[0]![1]);

    });
    let result: Awaited<ReturnType<typeof compile>> | undefined;

    const nativeSource = source.replace("export function test()", "async function test()").replace("__drain_microtasks();", "await aggregate;") + "\ntest();";

    save("native.js", nativeJS);

    let nativeError: string | null = null;

    catch (error) { nativeError = String(error); }
    let actual: unknown = null;
    let runtimeError: string | null = null;
    let imports: WebAssembly.ModuleImportDescriptor[] | null = null;
    if (result) {
      save("wat", result.wat);
      save("wasm", result.binary);
      if (result.success) {
        try {
          const module = await WebAssembly.compile(new Uint8Array(result.binary));
          imports = WebAssembly.Module.imports(module);
          const instance = await WebAssembly.instantiate(module, {});
          actual = (instance.exports.test as () => number)();

    }
