// Compare expanded values while recording the actual graph's identity map.
// This is ONLY the A/B repair diagnostic, never the B/C acceptance comparison.
function identityDelta(left, right, rootCount, callableLocal) {
  function index(root) {
    const nodes = new Map();
    function visit(value) {
      if (!value || typeof value !== "object") return;
      if (!Array.isArray(value) && Object.hasOwn(value, "id")) {
        assert(!nodes.has(value.id), "duplicate graph definition");
        nodes.set(value.id, value);
      }
      Object.values(value).forEach(visit);
    }
    visit(root);
    return nodes;
  }
  const ln = index(left), rn = index(right);
  const forward = new Map(), backward = new Map(), visited = new Set();
  const link = (map, key, value) => {
    if (!map.has(key)) map.set(key, new Set());
    map.get(key).add(value);
  };
  function walk(a, b, path) {
    if (!a || !b || typeof a !== "object" || typeof b !== "object") {
      assert.deepEqual(a, b, `A/B value mismatch at ${path}`);
      return;
    }
    if (Object.hasOwn(a, "ref")) { assert(ln.has(a.ref)); a = ln.get(a.ref); }
    if (Object.hasOwn(b, "ref")) { assert(rn.has(b.ref)); b = rn.get(b.ref); }
    const aNode = Object.hasOwn(a, "id"), bNode = Object.hasOwn(b, "id");
    assert.equal(aNode, bNode, `A/B node mismatch at ${path}`);
    if (aNode) {
      link(forward, a.id, b.id); link(backward, b.id, a.id);
      const pair = `${a.id}:${b.id}`;
      if (visited.has(pair)) return;
      visited.add(pair);
    }
    assert.equal(Array.isArray(a), Array.isArray(b), `A/B array mismatch at ${path}`);
    const ak = Object.keys(a).filter((key) => key !== "id");
    const bk = Object.keys(b).filter((key) => key !== "id");
    assert.deepEqual(ak, bk, `A/B ordered properties mismatch at ${path}`);
    for (const key of ak) walk(a[key], b[key], `${path}/${key}`);
  }
  walk(left, right, "root");
  const splits = [...forward].filter(([, ids]) => ids.size > 1).map(([oldId, ids]) => ({ oldId, newIds: [...ids], oldNode: ln.get(oldId), newNodes: [...ids].map((id) => rn.get(id)) }));
  const merges = [...backward].filter(([, ids]) => ids.size > 1).map(([newId, ids]) => ({ newId, oldIds: [...ids] }));
  // Preserve the measured raw identity delta even if its acceptance check fails.
  console.log(JSON.stringify({ kind: "AB-raw-identity-delta", rootCount, callableLocal, splits, merges }));
  assert.equal(splits.length, 2, "A/B must demonstrate exactly TWO actual shared-object splits");
  assert.deepEqual(merges, [], "A/B must not merge instruction identities");
  const expected = [
    [["op", "i32.const"], ["value", 1]],
    [["op", "local.set"], ["index", callableLocal]],
  ];
  assert.deepEqual(splits.map((split) => split.oldNode.entries), expected, "only resolve's two match leaves may split");
  for (const split of splits) {
    assert.equal(split.newIds.length, rootCount, "each shared leaf must split once per actual root");
    for (const node of split.newNodes) assert.deepEqual(node.entries, split.oldNode.entries);
  }
  return { splits, merges, expectedLeavesOnly: true, expandedValuesEqual: true };
}

const dirs = { A: aDir, B: bDir, C: cDir };

const rows = [];
for (const method of ["all", "race"]) {
  const records = {};
  for (const [arm, dir] of Object.entries(dirs)) {
    const row = JSON.parse(read(dir, `${method}.receipt.json`));
    records[arm] = row;
    assert.equal(row.arm, arm);
    assert.equal(row.method, method);
    assert.deepEqual(row.basis, { ...sharedBasis, [drivePath]: armBasis[arm][0], [protocolPath]: armBasis[arm][1] }, "exact complete pinned basis");
    assert.equal(row.sourceHash, sha(read(dir, `${method}.source.ts`)));
    assert.equal(row.binaryHash, sha(read(dir, `${method}.wasm`)));
    assert.deepEqual(row.observerErrors, []);
    assert.equal(row.hits.length, 1);
    assert.equal(row.hits[0].method, method);
    assert.equal(row.hits[0].arg, "input");
    assert.equal(row.hits[0].captured, true);
    assert.equal(row.hits[0].observable, true);
    assert(row.hits[0].entryBoundTypeIdx >= 0);
    assert(row.hits[0].entryBindProvider !== null);
    const observation = row.hits[0].observation;
    assert(observation.rootCount > 1);
    assert.equal(observation.rootCount, observation.distinctRoots);
    assert(observation.arms.some((item) => item.typeIdx === row.hits[0].boundTypeIdx));
    assert.equal(observation.uniqueLeaves, arm === "A" ? 2 : 2 * observation.rootCount);
    assert.equal(row.success, true);
    for (const key of ["compileError", "runtimeError", "nativeError"]) assert.equal(row[key], null);
    assert.deepEqual(row.imports, []);
    assert.equal(row.nativeActual, method === "all" ? 12 : 1);
    assert.equal(row.actual, row.nativeActual);
    assert.notDeepEqual(read(dir, `${method}.entry-0-before.json`), read(dir, `${method}.entry-0-after.json`), "non-vacuous emission delta");
  }
  // The ONLY B/C exclusions: already verified arm and drive/protocol hashes.
  function comparable(row) {
    const { arm, basis, ...rest } = row;
    const { [drivePath]: drive, [protocolPath]: protocol, ...shared } = basis;
    return { ...rest, basis: shared };
  }
  assert.deepEqual(comparable(records.B), comparable(records.C), `${method}: strict B/C receipts`);
  const suffixes = ["source.ts", "native.js", "entry-0-before.json", "entry-0-after.json", "wat", "wasm"];
  for (const suffix of suffixes) assert.deepEqual(read(bDir, `${method}.${suffix}`), read(cDir, `${method}.${suffix}`), `${method}: strict raw B/C ${suffix}`);
  for (const suffix of suffixes.filter((suffix) => suffix !== "entry-0-after.json")) assert.deepEqual(read(aDir, `${method}.${suffix}`), read(bDir, `${method}.${suffix}`), `${method}: A/B ${suffix}`);
  const a = records.A.hits[0].observation, b = records.B.hits[0].observation;
  assert.equal(a.rootCount, b.rootCount);
  const values = (observation) => observation.arms.map(({ typeIdx, leafValues }) => ({ typeIdx, leafValues }));
  assert.deepEqual(values(a), values(b));
  const delta = identityDelta(JSON.parse(read(aDir, `${method}.entry-0-after.json`)), JSON.parse(read(bDir, `${method}.entry-0-after.json`)), a.rootCount, records.A.hits[0].resolveLocals.callable[0][1]);
  // identityDelta above has independently proved exactly the two permitted
  // physical match-leaf splits. Check the receipt's identity observations too,
  // BEFORE omitting them from the A/B residual comparison.
  for (const [label, observation] of [["A", a], ["B", b]]) {
    assert.equal(observation.arms.length, observation.rootCount);
    const expectedIds = observation.arms.map((_, index) => label === "A" ? [0, 1] : [2 * index, 2 * index + 1]);
    assert.deepEqual(observation.arms.map((item) => item.leafIds), expectedIds, `${method}: ${label} actual match-leaf identities`);
    assert.equal(observation.uniqueLeaves, label === "A" ? 2 : 2 * observation.rootCount);
  }
  // Preserve EVERY remaining receipt field, including errors on successful
  // compiles, all hit metadata, local declarations, and future added fields.
  // comparable removes only the already verified arm/drive/protocol provenance.
  // These two identity fields are the sole additional A/B exclusions.
  function abResidual(row) {
    const residual = comparable(row);
    return {
      ...residual,
      hits: residual.hits.map((hit) => {
        const { uniqueLeaves, ...observation } = hit.observation;
        return {
          ...hit,
          observation: {
            ...observation,
            arms: observation.arms.map(({ leafIds, ...remaining }) => remaining),
          },
        };
      }),
    };
  }
  assert.deepEqual(abResidual(records.A), abResidual(records.B), `${method}: full A/B residual receipts`);
  rows.push({ method, roots: a.rootCount, AB: delta, BC: { rawGraphsEqual: true, watEqual: true, binaryEqual: true, receiptsEqualExceptVerifiedProvenance: true } });
}
