var C2;
class C { constructor() { C2 = C; } }
new C();
assert.sameValue(C2 === C, true, 'ctor-written var equals class');
var C3 = null;
class D { constructor() { C3 = D; } }
new D();
assert.sameValue(C3 === D, true, 'ctor-written null-init var equals class');
