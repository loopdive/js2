var s = Symbol();
class C { a() {} [s]() {} }
assert.compareArray(Object.getOwnPropertyNames(C.prototype), ['constructor', 'a']);
assert.sameValue(typeof new C()[s], 'function', 'symbol-keyed method reachable');
