class Base { constructor() {} }
var exn;
class Sub extends Base { constructor() { try { this.p = 3; } catch (e) { exn = e; } super(); } }
new Sub();
assert.sameValue(exn instanceof ReferenceError, true, 'this before super -> ReferenceError');
