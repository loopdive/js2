class Base1 {}
var threw = false;
try { Base1(); } catch (e) { threw = e instanceof TypeError; }
assert.sameValue(threw, true, 'module-scope call');
var threw2 = false;
(function() { try { Base1(); } catch (e) { threw2 = e instanceof TypeError; } })();
assert.sameValue(threw2, true, 'nested-scope call');
