var n = -1, a0;
class Base { constructor() { n = arguments.length; a0 = arguments[0]; } }
new Base(1, 2);
assert.sameValue(n, 2, 'arguments.length in class ctor');
assert.sameValue(a0, 1, 'arguments[0]');
