var caught;
class Parent { constructor() {} }
class Child extends Parent { constructor() { super(); try { super(); } catch (err) { caught = err; } } }
new Child();
assert.sameValue(typeof caught, 'object');
assert.sameValue(caught instanceof ReferenceError, true);
