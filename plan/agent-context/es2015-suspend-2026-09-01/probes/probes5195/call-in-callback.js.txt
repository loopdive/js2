class Base1 {}
function run(fn) { try { fn(); } catch (e) { return e instanceof TypeError; } return false; }
assert.sameValue(run(function() { Base1(); }), true, 'class call inside callback');
assert.sameValue(run(function() { return Base1(); }), true, 'class call inside callback (returned)');
