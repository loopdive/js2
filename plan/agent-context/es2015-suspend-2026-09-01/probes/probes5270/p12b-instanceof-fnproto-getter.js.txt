var called = 0;
Object.defineProperty(Function.prototype, 'prototype', { get: function() { called++; return Array.prototype; } });
assert.sameValue([] instanceof Function.prototype, true, 'walk reaches Array.prototype');
assert.sameValue(called, 1, 'getter called once: ' + called);
