function ID(x) { return x; }
var sym1 = Symbol(); var sym2 = Symbol();
var object = { a: 'A', [sym1]: 'B', c: 'C', [ID(sym2)]: 'D' };
assert.sameValue(object[sym1], 'B', 'sym1=' + String(object[sym1]));
assert.sameValue(object[sym2], 'D', 'sym2=' + String(object[sym2]));
assert.sameValue(Object.getOwnPropertySymbols(object).length, 2, 'symbols=' + Object.getOwnPropertySymbols(object).length);
