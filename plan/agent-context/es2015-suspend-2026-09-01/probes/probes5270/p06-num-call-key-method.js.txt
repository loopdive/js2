function ID(x) { return x; }
var object = { a() { return 'A'; }, [1]() { return 'B'; }, [ID(2)]() { return 'D'; } };
assert.sameValue(object[1](), 'B');
assert.sameValue(object[2](), 'D');
assert.sameValue(Object.getOwnPropertyNames(object).join(), '1,2,a', 'names=' + Object.getOwnPropertyNames(object).join());
