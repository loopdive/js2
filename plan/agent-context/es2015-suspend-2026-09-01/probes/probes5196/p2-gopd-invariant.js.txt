// §10.5.5 step 11: trap returns undefined for a non-configurable target prop → TypeError
var target = {};
Object.defineProperty(target, "attr", { configurable: false, value: 1 });

var threw = false;
try { Object.getOwnPropertyDescriptor(p, "attr"); } catch (e) { threw = e instanceof TypeError; }
if (!threw) throw new Error("no TypeError from gopd invariant");
