// Compile ONE file exactly like the standalone lane (src/index.ts compile, target "standalone")
// and print both the compile result's `imports` and the wasm module's real import list.
//   npx tsx .tmp/es2015/probes5267/imports-of.mts <file.js>
import { readFileSync } from "node:fs";
import { compile } from "../../../src/index.js";
const src = readFileSync(process.argv[2]!, "utf-8");
const r = await compile(src, { target: "standalone" } as never);
console.log("success:", (r as { success?: boolean }).success);
const errs = ((r as { errors?: { severity: string; message: string }[] }).errors ?? []).filter((e) => e.severity === "error");
console.log("errors:", JSON.stringify(errs.slice(0, 3).map((e) => e.message.slice(0, 160))));
console.log("result.imports:", JSON.stringify((r as { imports?: unknown[] }).imports));
const bin = (r as { binary?: Uint8Array }).binary;
if (bin) {
  const mod = new WebAssembly.Module(bin);
  console.log("module imports:", JSON.stringify(WebAssembly.Module.imports(mod).map((i) => `${i.kind}:${i.module}::${i.name}`)));
}
