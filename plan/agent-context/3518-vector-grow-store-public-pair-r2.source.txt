function tree(root: string, directory: string): [string, string][] {
  const result: [string, string][] = [];
  for (const entry of readdirSync(join(root, directory), { withFileTypes: true }).sort((a, b) =>
    a.name.localeCompare(b.name),
  )) {
    const path = join(directory, entry.name);
    if (entry.isDirectory()) result.push(...tree(root, path));
    else if (entry.isFile()) result.push([path, sha(readFileSync(join(root, path)))]);
  }
  return result;
}

function snapshot(root: string) {
  const files = tree(root, "src");
  return {
    root,
    head: execFileSync("git", ["-C", root, "rev-parse", "HEAD"], { encoding: "utf8" }).trim(),
    dirty: execFileSync("git", ["-C", root, "status", "--porcelain", "--untracked-files=all", "--", "src"], {
      encoding: "utf8",
    }),
    files,
    sha256: sha(JSON.stringify(files)),
  };
}

}

function data(value: unknown): unknown {
  const seen = new Map<object, number>();
  function visit(v: any): any {
    if (v === undefined) return { $undefined: true };
    if (typeof v === "number" && !Number.isFinite(v)) return { $number: String(v) };
    if (typeof v === "bigint") return { $bigint: String(v) };
    if (v === null || typeof v !== "object") return v;
    if (seen.has(v)) return { $ref: seen.get(v) };
    seen.set(v, seen.size);
    if (v instanceof Error)
      return Object.fromEntries(
        [...new Set(["name", "message", "stack", "cause", ...Object.getOwnPropertyNames(v)])].map((k) => [
          k,
          visit((v as Error & Record<string, unknown>)[k]),
        ]),
      );
    if (Array.isArray(v)) return v.map(visit);
    if (v instanceof Map) return { $map: [...v].map(([k, w]) => [visit(k), visit(w)]) };
    if (v instanceof Set) return { $set: [...v].map(visit) };
    return Object.fromEntries(Object.keys(v).map((k) => [k, visit(v[k])]));
  }
  return visit(value);
}

async function settle<T>(promise: Promise<T>, phase: string): Promise<T> {
  let rejectIdle!: (error: Error) => void;
  const idle = new Promise<never>((_, reject) => {
    rejectIdle = reject;
  });
  const onIdle = () => {
    const error = Object.assign(new Error("unsettled recorder await: " + phase), {
      code: "ERR_RECORDER_UNSETTLED_AWAIT",
      phase,
    });
    rejectIdle(error);
  };
  process.once("beforeExit", onIdle);
  try {
    return await Promise.race([promise, idle]);
  } finally {
    process.removeListener("beforeExit", onIdle);
  }
}

function artifact(result: CompileResult) {
  const bytes = result.binary;
  const module = new WebAssembly.Module(bytes);
  return {
    binary: Buffer.from(bytes).toString("base64"),
    wat: result.wat,
    imports: WebAssembly.Module.imports(module),
    exports: WebAssembly.Module.exports(module),
    resourceOrder: result.wat
      .split("\n")
      .filter((line) => /^ {2}\((?:type|import|global|func|export|elem|data)\b/.test(line)),
    units: data({
      compiled: result.irCompiledFuncs,
      outcomes: result.irOutcomes,
      postClaimErrors: result.irPostClaimErrors,
    }),
    stringPool: data(result.stringPool),
  };
}

// Extract exact balanced WAT forms, retaining all contents and coordinates.
// The scanner handles strings, line comments and nested WAT block comments.
function formAt(wat: string, start: number) {
  let depth=0, quoted=false, escaped=false, line=false, block=0;
  for(let i=start;i<wat.length;i++) {
    const c=wat[i], next=wat[i+1];
    if(line) { if(c === "\n") line=false; continue; }
    if(block) {
      if(c==="(" && next===";") {block++;i++;}
      else if(c===";" && next===")") {block--;i++;}
      continue;
    }
    if(quoted) {if(escaped)escaped=false;else if(c==="\\")escaped=true;else if(c==='"')quoted=false;continue;}
    if(c===";" && next===";"){line=true;i++;continue;}
    if(c==="(" && next===";"){block++;i++;continue;}
    if(c==='"'){quoted=true;continue;}
    if(c==="(")depth++;
    if(c===")" && --depth===0)return {start,end:i+1,text:wat.slice(start,i+1)};
  }
  throw Error("incomplete WAT form at "+start);
}
function paths(wat: string) {
  const allTypes=[...wat.matchAll(/\(type \$([^\s()]+)/g)].map(m=>({name:m[1],...formAt(wat,m.index!)}));
  const allFunctions=[...wat.matchAll(/^  \(func \$([^\s()]+)/gm)].map(m=>m[1]);
  const helpers=[...wat.matchAll(/\(func \$(__vec_elem_set_\d+)\b/g)].map(match=>{
    const form=formAt(wat,match.index!);
    const arrayIndices=[...form.text.matchAll(/\barray\.set (\d+)\b/g)].map(m=>Number(m[1]));
    assert.equal(arrayIndices.length,1,"one actual array store in grow/store helper");
    const backing=allTypes[arrayIndices[0]];
    assert(backing,"resolved actual numeric array coordinate");
    const elementMatch=backing.text.match(/\(array \(mut (f64|externref|i32|i64|f32|i8|i16)\)\)/);
    const element=elementMatch?.[1] ?? "reference";
    const fill=form.text.includes("f64.reinterpret_i64")?"f64-hole":form.text.includes("global.get")?"hole-global":"default";
    const functionIndex=allFunctions.indexOf(match[1]);
    assert(functionIndex>=0,"actual defined function coordinate");
    // Every selected module is independently required to have zero imports;
    // therefore its definition order is its complete numeric function space.
    const callCount=[...wat.matchAll(new RegExp("\\bcall "+functionIndex+"(?=\\s|\\))","g"))].length;
    return {name:match[1],...form,sha256:sha(form.text),element,fill,functionIndex,arrayTypeIndex:arrayIndices[0],backing,callCount,category:fill+"-"+element};
  });
  const types=[...wat.matchAll(/\(type \$(__vec_(?:base|i8_byte|i32_byte|externref|f64)|__arr_(?:externref|f64)|__holey_array)\b/g)]
    .map(match=>({name:match[1],...formAt(wat,match.index!)}));
  const packed=types.find(t=>t.name==="__vec_i8_byte"), buffer=types.find(t=>t.name==="__vec_i32_byte");
  const branded=!!packed && !!buffer && packed.text.includes("(sub final ") && buffer.text.includes("(sub $") && !buffer.text.includes("(sub final ");
  return {helpers,types,branded};
}
async function measure(api: any, fixture: (typeof matrix)[number]) {
  let phase="compile";
  const row: any={id:fixture.id,fixture,compilations:[],executions:[],kind:"pending"};
  try {
    const result=await settle(api.compile(fixture.source,fixture.options),"compile:"+fixture.id);
    const compilation:any={source:fixture.source,options:fixture.options,success:result.success,errors:data(result.errors),
      artifact:{binary:Buffer.from(result.binary).toString("base64"),wat:result.wat}};
    row.compilations.push(compilation);
    if(!result.success){row.kind="refused";row.phase=phase;return row;}
    phase="artifact";
    compilation.artifact=artifact(result);
    compilation.artifact.completeResourceOrder=result.wat.split("\n").filter((line:string)=>
      /^\s+\((?:rec|type|import|global|func|tag|table|memory|export|elem|data)(?:\s|\))/.test(line));
    row.pathEvidence=paths(result.wat);
    assert(result.wat.length>0);
    assert(result.binary.length>8);
    assert.deepEqual(compilation.artifact.imports,[],"standalone zero imports");
    assert.deepEqual(result.imports,[],"closed declared import census");
    phase="instantiate";
    const bytes=Buffer.from(result.binary); // preserve exact input to actual instantiation
    const input=bytes.toString("base64");
    assert.equal(input,compilation.artifact.binary);
    const execution:any={instantiationInputBinary:input,instantiatedBinary:null,values:[]};
    row.executions.push(execution);
    const {instance}=await settle(WebAssembly.instantiate(bytes,{}),"instantiate:"+fixture.id);
    execution.instantiatedBinary=input;
    phase="execute";
    assert.equal(typeof instance.exports.run,"function");
    for(let ordinal=0;ordinal<2;ordinal++) {
      const value=(instance.exports.run as Function)();
      execution.values.push({export:"run",args:[],ordinal,value:data(value)});
      assert.deepEqual(data(value),fixture.expected,"independent expected result: "+fixture.id);
    }
    row.kind="executed"; row.phase="completed";
  } catch(error) {row.kind="threw";row.phase=phase;row.error=data(error);}
  return row;
}

function differences(a:any,b:any,path="$",out:any[]=[]): any[] {
  if(Object.is(a,b))return out;
  if(!a || !b || typeof a!=="object" || typeof b!=="object" || Array.isArray(a)!==Array.isArray(b)) {
    out.push({path,baseline:a,candidate:b}); return out;
  }
  for(const k of new Set([...Object.keys(a),...Object.keys(b)]))differences(a[k],b[k],path+"."+k,out);
  return out;
}

  const mustReject=(name:string,operation:()=>unknown)=>{assert.throws(operation);negatives.push({name,rejected:true});};
  const mutate=(fn:(x:any)=>void)=>{const x=structuredClone(after);fn(x);return x;};
  mustReject("drop-row",()=>validateReport(mutate(x=>x.rows.pop()),"candidate",expected));
  mustReject("wrong-root",()=>validateReport(before,"candidate",expected));
  const executedIndex=after.rows.findIndex((r:any)=>r.kind==="executed");
  assert(executedIndex>=0,"positive executing comparator control");
  mustReject("alter-instantiated-byte",()=>validateReport(mutate(x=>x.rows[executedIndex].executions[0].instantiatedBinary+="AA"),"candidate",expected));
  mustReject("alter-value",()=>validateReport(mutate(x=>x.rows[executedIndex].executions[0].values[0].value=99999),"candidate",expected));
  const changed=mutate(x=>x.rows[executedIndex].compilations[0].errors.push({message:"changed outcome"}));
  assert(differences(before.rows,changed.rows).length>0);negatives.push({name:"changed-outcome-not-equal",rejected:true});
  const report={schema:"vector-grow-store-public-pair-v2",preservationOK,acceptanceOK,baseline:a,candidate:b,denominator:{pairs:14,rows:28,expectedCalls:56},terminals,rawDifferences,negativeControls:negatives,expected};
  writeFileSync(join(directory,"comparison.json"),JSON.stringify(report,null,2));
  assert.deepEqual(sourceCensus(),expected);
  console.log(JSON.stringify({stage:"comparison",preservationOK,acceptanceOK,baseline:a,candidate:b,rawDifferenceCount:rawDifferences.length,path:join(directory,"comparison.json")}));
  if(!acceptanceOK)process.exitCode=1;
}
