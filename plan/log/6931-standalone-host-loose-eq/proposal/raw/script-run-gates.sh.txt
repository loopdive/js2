#!/usr/bin/env bash
# Usage: bash .tmp/run-gates.sh <label> — each gate run bare, exit code recorded.
cd "$(dirname "$0")/.."
label="$1"
out=".tmp/6931-out/raw"
summary="$out/${label}-gates.tsv"
printf "gate\texit\n" > "$summary"
run() {
  local name="$1"; shift
  "$@" > "$out/${label}-gate-${name}.txt" 2>&1
  local rc=$?
  printf "%s\t%s\n" "$name" "$rc" >> "$summary"
}
run typecheck npm run -s typecheck
run loc-budget node scripts/check-loc-budget.mjs
run func-budget node scripts/check-func-budget.mjs
run coercion-sites node scripts/check-coercion-sites.mjs
run oracle-ratchet npm run -s check:oracle-ratchet
run dead-exports npm run -s check:dead-exports
run ir-fallbacks pnpm run check:ir-fallbacks
run compiler-boundaries node scripts/check-compiler-boundaries.mjs --mode inventory --base dbf5b4f7
cat "$summary"
