// Usage: npx tsx .tmp/sha-corpus.mts <out.tsv>
// Compiles every playground example + the #6931 probe sources on gc and standalone
// through the public compile() and records sha256 + size per binary.
import { createHash } from "crypto";
import { readFileSync, writeFileSync, readdirSync, statSync } from "fs";
import { join, relative } from "path";
import { compile } from "../src/index.js";

const root = process.cwd();
function walk(d: string, out: string[] = []): string[] {
  for (const e of readdirSync(d)) {
    const p = join(d, e);
    if (statSync(p).isDirectory()) walk(p, out);
    else if (p.endsWith(".ts")) out.push(p);
  }
  return out;
}
const files: [string, string][] = walk(join(root, "website/playground/examples"))
  .sort()
  .map((p) => [relative(root, p), readFileSync(p, "utf8")]);
const PROBES: Record<string, string> = {
  "probe:boundary": `export function eqUsed(o: any): number { return o == 1 ? 1 : 0; }
export function eqStr(o: any): number { return o == "1" ? 1 : 0; }
export function nativeObj(): number { const o: any = { a: 1 }; return o == 1 ? 1 : 0; }
export function nativeStr(): number { const o: any = "1"; return o == 1 ? 1 : 0; }
export function nativeNum(): number { const o: any = 1; return o == 1 ? 1 : 0; }
`,
  "probe:native": `function eqUsed(o: any): number { return o == 1 ? 1 : 0; }
export function ident(o: any): any { return o; }
export function viaStr(): number { return eqUsed("1"); }
export function viaBool(): number { return eqUsed(true); }
export function viaObj(): number { return eqUsed({ valueOf() { return 1; } }); }
export function viaUndef(): number { return eqUsed(undefined); }
export function exported(o: any): number { return eqUsed(o); }
`,
  "probe:wat": `export function eqUsed(o: any): number { return o == 1 ? 1 : 0; }\n`,
  "probe:no-eq": `export function add(a: number, b: number): number { return a + b; }\n`,
  "probe:strict-only": `export function s(o: any): number { return o === 1 ? 1 : 0; }\n`,
};
for (const [k, v] of Object.entries(PROBES)) files.push([k, v]);
const rows: string[] = ["name\ttarget\tok\tsize\tsha256"];
for (const [name, src] of files) {
  for (const target of ["gc", "standalone"] as const) {
    let r: any;
    try {
      r = await compile(src, { fileName: "input.ts", ...(target === "standalone" ? { target } : {}) });
    } catch (e: any) {
      rows.push(`${name}\t${target}\tTHROW\t0\t${String(e?.message).slice(0, 60)}`);
      continue;
    }
    if (!r.success) {
      rows.push(`${name}\t${target}\tFAIL\t0\t-`);
      continue;
    }
    const h = createHash("sha256").update(r.binary).digest("hex");
    rows.push(`${name}\t${target}\tok\t${r.binary.length}\t${h}`);
  }
}
writeFileSync(process.argv[2]!, rows.join("\n") + "\n");
console.log(`wrote ${rows.length - 1} rows`);
