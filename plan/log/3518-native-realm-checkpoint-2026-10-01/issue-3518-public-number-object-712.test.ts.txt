import { afterEach, describe, expect, it } from "vitest";
import { readFileSync } from "node:fs";
import { createHash } from "node:crypto";
import { runInNewContext } from "node:vm";
import ts from "typescript";
import { compile } from "../src/index.js";
import { buildImports } from "../src/runtime.js";
import { prepareWholeIrProgram } from "../src/ir/program-preparation.js";
import { decodePreparedIrProgram, encodePreparedIrProgram } from "../src/ir/program-codec.js";
import { acceptPreparedIrProgram, emitAcceptedIrProgram } from "../src/ir/program-consumer.js";
import { emitBinary } from "../src/emit/binary.js";
import { requireProgram, sourceInput } from "./helpers/typed-program-fixtures.js";
import { replayOptions } from "./helpers/ir-whole-program-replay.js";

afterEach(() => new Promise<void>((resolve) => setTimeout(resolve, 0)));

const source = readFileSync("tests/fixtures/issue-3518-native-object-access-712.ts.txt", "utf8");
function oracle(): number {
  const exports: { run?: () => number } = {};
  runInNewContext(
    ts.transpileModule(source, { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022 } })
      .outputText,
    { exports },
    { timeout: 1000 },
  );
  if (!exports.run) throw new Error("original fixture lost its public run export");
  return exports.run();
}
const policy = {
  backend: "wasmgc",
  target: "standalone",
  numberBoundary: { box: "native", unbox: "native" },
  booleanBoundary: { box: "native", unbox: "native" },
  stringConst: { storage: "native" },
  stringEq: { eq: "native" },
} as const;

describe("unchanged712 legacy and public native Number parity", () => {
  it("preserves the original source and observes the native712 oracle", () => {
    expect(createHash("sha256").update(source).digest("hex")).toBe(
      "c0550b99175c0eb61afa7d5d110a287f3fe90e9fc8e581c6d58a19fe0a971ba9",
    );
    expect(oracle()).toBe(712);
  });
  for (const target of ["host", "standalone"] as const)
    for (const utf8Storage of [false, true]) {
      it(`preserves explicit legacy execution, target=${target}, utf8=${utf8Storage}`, async () => {
        const result = await compile(source, { target, experimentalIR: false, utf8Storage });
        expect(result.success, JSON.stringify(result.errors)).toBe(true);
        const module = new WebAssembly.Module(result.binary as BufferSource);
        if (target === "standalone") expect(WebAssembly.Module.imports(module)).toEqual([]);
        for (let fresh = 0; fresh < 2; fresh++) {
          const imports = target === "host" ? buildImports(result.imports, undefined, result.stringPool) : undefined;
          const instance = new WebAssembly.Instance(module, imports ?? {});
          imports?.setInstance?.(instance);
          const run = instance.exports.run as () => number;
          expect(run()).toBe(oracle());
          expect(run()).toBe(712);
        }
      });
    }
  for (const decoded of [false, true])
    for (const utf8Storage of [false, true]) {
      it(`executes through actual public acceptance/emission, decoded=${decoded}, utf8=${utf8Storage}`, () => {
        const original = requireProgram(
          prepareWholeIrProgram({
            ...sourceInput({ "./entry.ts": source }),
            policy,
            runtimePolicies: [policy],
            nativeStringValueProjection: "standalone-native",
          }),
        );
        const bytes = encodePreparedIrProgram(original);
        const program = decoded ? decodePreparedIrProgram(bytes) : original;
        expect(encodePreparedIrProgram(program)).toBe(bytes);
        const accepted = acceptPreparedIrProgram(program, { ...replayOptions("wasmgc", "standalone"), utf8Storage });
        expect(accepted.kind, JSON.stringify(accepted)).toBe("accepted");
        if (accepted.kind !== "accepted") throw new Error(JSON.stringify(accepted));
        const emitted = emitAcceptedIrProgram(accepted);
        expect(emitted.module.imports).toEqual([]);
        const module = new WebAssembly.Module(emitBinary(emitted.module) as BufferSource);
        expect(WebAssembly.Module.imports(module)).toEqual([]);
        for (let fresh = 0; fresh < 2; fresh++) {
          const exports = new WebAssembly.Instance(module).exports as unknown as { run: () => number };
          expect(exports.run()).toBe(oracle());
          expect(exports.run()).toBe(712);
        }
      });
    }
});
