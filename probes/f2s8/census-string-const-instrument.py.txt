# Every patch asserts an exact occurrence count so a drifted anchor fails loudly.
def patch(path, old, new, count=1):
    s = open(path, encoding="utf8").read()
    n = s.count(old)
    assert n == count, f"{path}: expected {count} of {old[:80]!r}, found {n}"
    open(path, "w", encoding="utf8").write(s.replace(old, new))
    print("patched", path, old[:50].replace("\n", "\\n"))
