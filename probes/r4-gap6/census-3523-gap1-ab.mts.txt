// Modes:
//   base  : as on origin/main e36225ee (call-free gate as landed)
//   SKIP2 : pass 2 never runs → the emitted body is pass 1's (fixup-maintained) body
//   SKIP1 : pass 1 never runs, pass 2 forced → the body compiles ONCE, after all function bodies

const LANES = [
  { name: "host-start", opts: { target: "gc" as const } },
  { name: "host-deferred", opts: { target: "gc" as const, deferTopLevelInit: true } },
  { name: "standalone", opts: { target: "standalone" as const } },

const MODES = [
  { name: "base", env: {} },
  { name: "SKIP2", env: { JS2WASM_PROBE_SKIP_PASS2: "1" } },
  { name: "SKIP1", env: { JS2WASM_PROBE_SKIP_PASS1: "1", JS2WASM_TEST_FORCE_MODULE_INIT_PASS2: "1" } },
];
const ALL_ENV = ["JS2WASM_PROBE_SKIP_PASS1", "JS2WASM_PROBE_SKIP_PASS2", "JS2WASM_TEST_FORCE_MODULE_INIT_PASS2"];
function setMode(env: Record<string, string>): void {
  for (const k of ALL_ENV) Reflect.deleteProperty(process.env, k);
  Object.assign(process.env, env);
}

function passCensus(): { p1: number; p2: number } {
  const rows = getCompileProfile();
  const calls = (suffix: string): number =>
    rows.filter((r: any) => r.path === suffix || r.path.endsWith(`/${suffix}`)).reduce((s: number, r: any) => s + r.calls, 0);
  return { p1: calls("module-init-pass1"), p2: calls("module-init-pass2") };
}

async function instantiate(result: any, lane: (typeof LANES)[number]): Promise<Record<string, any>> {
  if (lane.opts.target === "standalone" || lane.opts.target === "wasi") {
    const mod = await WebAssembly.compile(result.binary);
    const imports: Record<string, Record<string, unknown>> = {};
    for (const imp of WebAssembly.Module.imports(mod)) {
      imports[imp.module] ??= {};
      if (imp.kind === "function") imports[imp.module]![imp.name] = () => 0;
      else if (imp.kind === "memory") imports[imp.module]![imp.name] = new WebAssembly.Memory({ initial: 1 });
      else if (imp.kind === "global") imports[imp.module]![imp.name] = new WebAssembly.Global({ value: "i32", mutable: false }, 0);
      else if (imp.kind === "table") imports[imp.module]![imp.name] = new WebAssembly.Table({ initial: 0, element: "anyfunc" });
    }
    const instance = await WebAssembly.instantiate(mod, imports);
    return instance.exports as Record<string, any>;
  }
  const imports = buildImports(result.imports, {}, result.stringPool);
  const { instance } = await instantiateWasm(result.binary, imports.env, imports.string_constants);
  imports.setInstance?.(instance);
  return instance.exports as Record<string, any>;
}

async function runCell(src: string, fileName: string, lane: (typeof LANES)[number], expect: unknown) {
  resetCompileProfile();
  const out: any = { success: false, p1: 0, p2: 0, sha: "", bytes: 0, runtime: "", errors: "", outcome: "" };
  try {
    const result: any = await compile(src, { fileName, ...lane.opts, skipSemanticDiagnostics: true, trackIrOutcomes: true, emitWat: false });
    const c = passCensus(); out.p1 = c.p1; out.p2 = c.p2;
    out.success = !!result.success;
    const rows = (result.irOutcomes ?? []).filter((o: any) => o.unitKind === "module-init");
    out.outcome = rows.length === 1 ? `${rows[0].kind}${rows[0].code ? "/" + rows[0].code : ""} L${rows[0].legacyBodyEmitted ? 1 : 0}I${rows[0].irBodyEmitted ? 1 : 0}` : `rows=${rows.length}`;
    out.errors = (result.errors ?? []).map((e: any) => String(e.message).replace(/\s+/g, " ").slice(0, 110)).join(" || ");
    if (result.success && result.binary) {
      out.bytes = result.binary.length;
      out.sha = createHash("sha1").update(Buffer.from(result.binary)).digest("hex").slice(0, 10);
      out.valid = WebAssembly.validate(result.binary);
      try {
        const ex = await instantiate(result, lane);
        if (lane.opts.deferTopLevelInit && typeof ex.__module_init === "function") ex.__module_init();
        const v = ex.read?.();
        out.runtime = `${String(v)}${v === expect ? " OK" : " MISMATCH"}`;
      } catch (e: any) { out.runtime = `RUNFAIL:${String(e?.message ?? e).slice(0, 80)}`; }
    }
  } catch (e: any) { out.errors = `THROW:${String(e?.message ?? e).slice(0, 140)}`; }
  return out;
}

const cells: any[] = [];
for (const shape of SHAPES) {
  for (const lane of LANES) {
    const row: any = { shape: shape.id, lane: lane.name };
    for (const mode of MODES) {
      setMode(mode.env);
      row[mode.name] = await runCell(shape.src, `ab-${shape.id}-${lane.name}.ts`, lane, shape.expect);
    }
    setMode({});
    cells.push(row);
  }
}

}

// ---- corpus section: playground + examples, host-start and standalone
function* walk(dir: string): Generator<string> {
  if (!fs.existsSync(dir)) return;
  for (const ent of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, ent.name);
    if (ent.isDirectory()) yield* walk(p);
    else if (ent.isFile() && /\.ts$/.test(ent.name) && !ent.name.endsWith(".d.ts")) yield p;
  }
}
const corpusFiles = [...walk(path.join(WT, "website/playground/examples")), ...walk(path.join(WT, "examples"))];
const corpus: any[] = [];
for (const f of corpusFiles) {
  const src = fs.readFileSync(f, "utf8");
  for (const lane of LANES.filter((l) => l.name === "host-start" || l.name === "standalone")) {
    const row: any = { file: path.relative(WT, f), lane: lane.name };
    for (const mode of MODES) {
      setMode(mode.env);
      resetCompileProfile();
      try {
        const result: any = await compile(src, { fileName: path.basename(f), ...lane.opts, skipSemanticDiagnostics: true, emitWat: false });
        const c = passCensus();
        row[mode.name] = { success: !!result.success, p1: c.p1, p2: c.p2, sha: result.binary ? createHash("sha1").update(Buffer.from(result.binary)).digest("hex").slice(0, 10) : "", bytes: result.binary?.length ?? 0, errs: (result.errors ?? []).length, firstErr: (result.errors ?? [])[0]?.message?.replace(/\s+/g, " ").slice(0, 100) ?? "" };
      } catch (e: any) { row[mode.name] = { success: false, p1: 0, p2: 0, sha: "", bytes: 0, errs: -1, firstErr: `THROW:${String(e?.message ?? e).slice(0, 100)}` }; }
    }
    setMode({});
    corpus.push(row);
  }
}
